Select

select reads rows from a table. * means all columns.

SELECT FROM table

mysql> SELECT * FROM mytable;
mysql> SELECT column1, column2 FROM mytable;

ORDER BY

ASC sorts from low to high, which is the default. DESC sorts from high to low.

mysql> SELECT * FROM mytable ORDER BY column1 ASC;
mysql> SELECT * FROM mytable ORDER BY column1 DESC;

WHERE

WHERE only returns the rows that match.

mysql> SELECT * FROM `mytable` WHERE id=1;

AND, OR and NOT

mysql> SELECT * FROM mytable WHERE column1 = 'a' AND column2 = 'b';
mysql> SELECT * FROM mytable WHERE column1 = 'a' OR column1 = 'b';
mysql> SELECT * FROM mytable WHERE NOT column1 = 'a';

IS NULL and IS NOT NULL

mysql> SELECT * FROM mytable WHERE column1 IS NULL;
mysql> SELECT * FROM mytable WHERE column1 IS NOT NULL;

LIMIT

LIMIT returns at most the given number of rows.

mysql> SELECT * FROM mytable WHERE column1 IS NOT NULL LIMIT 10;

MIN and MAX

mysql> SELECT MIN(id) FROM mytable;
mysql> SELECT MAX(id) FROM mytable;

Wildcard

LIKE matches a pattern. % means any characters, so '/%' matches everything that starts with a slash.

mysql> SELECT * FROM `mytable` WHERE column1 LIKE '/%';
mysql> SELECT * FROM `mytable` WHERE column1 NOT LIKE '/%';
