Create database

create database makes a new, empty database.

Create

mysql> create database mydatabase;

Give a user access

GRANT gives a user rights to the database. testdb.* means every table in testdb.

mysql> create database testdb;
mysql> GRANT ALL PRIVILEGES ON testdb.* TO 'user'@'localhost';
