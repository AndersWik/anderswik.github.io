Select distinct

SELECT DISTINCT works like select, but removes duplicates so every value is returned only once. ORDER BY, WHERE, AND, OR, NOT, IS NULL and LIMIT work the same way.

SELECT DISTINCT FROM table

mysql> SELECT DISTINCT column1 FROM mytable;

ORDER BY

mysql> SELECT DISTINCT column1 FROM mytable ORDER BY column1 ASC;
mysql> SELECT DISTINCT column1 FROM mytable ORDER BY column1 DESC;

WHERE

mysql> SELECT DISTINCT column1 FROM mytable WHERE column2 = 'a';
mysql> SELECT DISTINCT column1 FROM mytable WHERE column2 = 'a' AND column1 IS NOT NULL;
mysql> SELECT DISTINCT column1 FROM mytable WHERE column2 = 'a' OR column2 = 'b';
mysql> SELECT DISTINCT column1 FROM mytable WHERE NOT column2 = 'a';

LIMIT

mysql> SELECT DISTINCT column1 FROM mytable LIMIT 10;

Count distinct values

mysql> SELECT COUNT(DISTINCT column1) FROM mytable;
