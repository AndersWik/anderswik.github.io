Beamer

Beamer is a document class for presentations. Each slide is a frame.

Presentation

\documentclass{beamer}
\usetheme{Madrid}

\title{My talk}
\author{Anders Wik}
\date{\today}

\begin{document}

\begin{frame}
  \titlepage
\end{frame}

\begin{frame}{First slide}
  \begin{itemize}
    \item One
    \item Two
  \end{itemize}
\end{frame}

\end{document}

Step by step

\begin{frame}{Step by step}
  \begin{itemize}
    \item<1-> Shown first
    \item<2-> Shown second
    \item<3-> Shown third
  \end{itemize}
  \pause
  Shown last.
\end{frame}

Columns

\begin{frame}{Two columns}
  \begin{columns}
    \column{0.5\textwidth}
    Text on the left.
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth]{penguin}
  \end{columns}
\end{frame}

Blocks

\begin{block}{Title}
  A box.
\end{block}
\begin{alertblock}{Warning}
  A red box.
\end{alertblock}
\begin{exampleblock}{Example}
  A green box.
\end{exampleblock}

Themes

\usetheme{default}      % also Madrid, Berlin, Warsaw, metropolis
\usecolortheme{beaver}  % also dolphin, whale, seahorse
\setbeamertemplate{navigation symbols}{}   % hide the navigation icons

More about Beamer
