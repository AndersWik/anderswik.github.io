REST API

REST is a way to design web services around resources. Each resource has a URL, like /todos/5, and the HTTP method says what to do with it. The data is usually sent as JSON, and the status code tells how it went.

  • GET /todos lists all todos
  • POST /todos creates a todo
  • GET /todos/5 reads one todo
  • PATCH /todos/5 changes a todo
  • DELETE /todos/5 deletes a todo

The API

The whole API is one file. It stores the todos in SQLite, uses prepared statements for all values and validates the JSON it gets.

<?php
// public/api.php
declare(strict_types=1);

header('Content-Type: application/json; charset=utf-8');

function respond(int $status, mixed $data = null): never {
    http_response_code($status);
    if ($data !== null) {
        echo json_encode($data, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
    }
    exit;
}

function jsonBody(): array {
    $data = json_decode(file_get_contents('php://input'), true);
    if (!is_array($data)) {
        respond(400, ['error' => 'The body must be a JSON object']);
    }
    return $data;
}

$pdo = new PDO('sqlite:' . __DIR__ . '/../data/todos.sqlite', null, null, [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
]);
$pdo->exec('CREATE TABLE IF NOT EXISTS todos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    done INTEGER NOT NULL DEFAULT 0
)');

$method = $_SERVER['REQUEST_METHOD'];
$path = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH) ?: '/';

if ($path === '/todos') {
    if ($method === 'GET') {
        respond(200, $pdo->query('SELECT id, title, done FROM todos ORDER BY id')->fetchAll());
    }

    if ($method === 'POST') {
        $input = jsonBody();
        $title = is_string($input['title'] ?? null) ? trim($input['title']) : '';

        if ($title === '' || mb_strlen($title) > 200) {
            respond(422, ['error' => 'title is required and can be at most 200 characters']);
        }

        $stmt = $pdo->prepare('INSERT INTO todos (title) VALUES (:title)');
        $stmt->execute(['title' => $title]);
        $id = (int) $pdo->lastInsertId();

        header('Location: /todos/' . $id);
        respond(201, ['id' => $id, 'title' => $title, 'done' => 0]);
    }

    header('Allow: GET, POST');
    respond(405, ['error' => 'Method not allowed']);
}

if (preg_match('#^/todos/(\d+)$#', $path, $matches)) {
    $stmt = $pdo->prepare('SELECT id, title, done FROM todos WHERE id = :id');
    $stmt->execute(['id' => (int) $matches[1]]);
    $todo = $stmt->fetch();

    if ($todo === false) {
        respond(404, ['error' => 'Todo not found']);
    }

    if ($method === 'GET') {
        respond(200, $todo);
    }

    if ($method === 'PATCH') {
        $input = jsonBody();
        if (!is_bool($input['done'] ?? null)) {
            respond(422, ['error' => 'done must be true or false']);
        }

        $stmt = $pdo->prepare('UPDATE todos SET done = :done WHERE id = :id');
        $stmt->execute(['done' => (int) $input['done'], 'id' => $todo['id']]);
        $todo['done'] = (int) $input['done'];
        respond(200, $todo);
    }

    if ($method === 'DELETE') {
        $stmt = $pdo->prepare('DELETE FROM todos WHERE id = :id');
        $stmt->execute(['id' => $todo['id']]);
        respond(204);
    }

    header('Allow: GET, PATCH, DELETE');
    respond(405, ['error' => 'Method not allowed']);
}

respond(404, ['error' => 'Not found']);

never as return type means the function never returns, it always ends the script. It needs PHP 8.1.

Run and test

$ mkdir data
$ php -S localhost:8000 -t public public/api.php
$ curl -i http://localhost:8000/todos
$ curl -i -X POST -H "Content-Type: application/json" -d '{"title":"Buy milk"}' http://localhost:8000/todos
$ curl -i -X PATCH -H "Content-Type: application/json" -d '{"done":true}' http://localhost:8000/todos/1
$ curl -i -X DELETE http://localhost:8000/todos/1

Before going public

  • Require authentication, for example a secret API token in the Authorization header that is compared with hash_equals
  • Always use HTTPS
  • Limit how many requests a client can make
  • Log errors and return a general 500 message without details
