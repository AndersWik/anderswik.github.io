Calculator

We build a calculator: digits, + - X /, a decimal point, backspace, clear, and a % button that works like on a pocket calculator. The new idea in this tutorial is state: the calculator has to remember what you have typed so far.

Open the finished calculator

State, and drawing from state

The calculator remembers three things, kept as text:

firstNumber    "200"   the number before the operator
operator       "+"     one of + - X /
secondNumber   "20"    the number after the operator

Every button press changes the state, and then one function draws the display from the state. The display is never edited directly. That makes the code easy to follow: to know what is on the screen, you only have to look at three variables.

We keep the numbers as text while typing, so that 0. and 12.50 look right. They are turned into real numbers only when we calculate.

Step 3: The state and the display

let firstNumber = "";
let operator = "";
let secondNumber = "";
let showingResult = false;   // true right after "=" or "%"

const display = document.getElementById("output");
const buttons = document.querySelectorAll(".calc-button");

function updateDisplay() {
    display.value = firstNumber + operator + secondNumber;
}

The state uses let because it changes. The elements use const because they never change. querySelectorAll finds every element that matches a CSS selector, here all 19 buttons.

Step 4: The maths

function calculate(a, op, b) {
    if (op === "+") return a + b;
    if (op === "-") return a - b;
    if (op === "X") return a * b;
    if (op === "/") return a / b;
}

Percent works like on a pocket calculator. You type the total first, then the percentage:

//   200 X 20%  ->  40    (20% of 200)
//   200 + 20%  ->  240   (200 plus 20% of 200)
//   200 - 20%  ->  160   (200 minus 20% of 200)
//   200 / 20%  ->  1000  (200 divided by 0.2)
function calculatePercent(a, op, b) {
    const percentOfA = a * (b / 100);

    if (op === "+") return a + percentOfA;
    if (op === "-") return a - percentOfA;
    if (op === "X") return percentOfA;
    if (op === "/") return a / (b / 100);
}

Step 5: Showing a result

The result becomes the new first number, so you can keep calculating with it. Two things can go wrong. Dividing by zero gives Infinity, and broken input gives NaN. isFinite() is false for both:

// 0.1 + 0.2 is 0.30000000000000004 in JavaScript. 12 significant digits hide that.
function round(number) {
    return Number(number.toPrecision(12));
}

function showResult(result) {
    if (!isFinite(result)) {
        clearAll();
        display.value = "Error";
        return;
    }

    firstNumber = String(round(result));
    operator = "";
    secondNumber = "";
    showingResult = true;
    updateDisplay();
}

showingResult remembers that the display shows an answer. If the next key is a digit, we start a new calculation. If it is an operator, we keep calculating with the answer.

Step 6: Typing a digit

function typeDigit(digit) {
    if (showingResult) {
        clearAll();
    }
    // Before an operator is chosen we are typing the first number.
    let current = operator === "" ? firstNumber : secondNumber;

    // A number can only have one decimal point.
    if (digit === "." && current.includes(".")) {
        return;
    }
    // Write "0.5" instead of ".5".
    if (digit === "." && current === "") {
        digit = "0.";
    }

    current = current + digit;

    if (operator === "") {
        firstNumber = current;
    } else {
        secondNumber = current;
    }
    updateDisplay();
}

Step 7: Operators, equals and percent

When you press an operator after 200+20, we first calculate 200+20, like a pocket calculator does. Then we continue with the new operator:

function chooseOperator(newOperator) {
    if (firstNumber === "") {
        return;                     // we need a number first
    }
    if (secondNumber !== "") {
        pressEquals();
    }
    if (firstNumber === "") {
        return;                     // that calculation was an error, like 1/0
    }
    operator = newOperator;
    showingResult = false;
    updateDisplay();
}

function pressEquals() {
    if (secondNumber === "") {
        return;
    }
    showResult(calculate(Number(firstNumber), operator, Number(secondNumber)));
}

pressPercent() is the same as pressEquals() but uses calculatePercent. Clear sets all the state back to empty. Backspace removes the last thing typed: a digit of the second number first, then the operator, then digits of the first number. slice(0, -1) gives a string without its last character.

Step 8: One click handler for all buttons

Instead of 19 functions, one function reads data-val and decides what to do. event.currentTarget is the element the listener is on, which is the button:

function onButtonClick(event) {
    const value = event.currentTarget.dataset.val;

    if (value === "ac") {
        clearAll();
    } else if (value === "<") {
        backspace();
    } else if (value === "=") {
        pressEquals();
    } else if (value === "%") {
        pressPercent();
    } else if (["+", "-", "X", "/"].includes(value)) {
        chooseOperator(value);
    } else {
        typeDigit(value); // 0-9 or "."
    }
}

buttons.forEach(function (button) {
    button.addEventListener("click", onButtonClick);
});

  • 0.1 + 0.2 = gives 0.3.

  • 200 + 20 X shows 220X.
  • 1 / 0 = shows Error.
  • 1 . . 5 shows 1.5.

  <script>
    /*
     * A simple calculator.
     *
     * The calculator remembers three things (its "state"):
     *
     *   firstNumber   the number typed before the operator    e.g. "200"
     *   operator      one of "+", "-", "X", "/"                e.g. "+"
     *   secondNumber  the number typed after the operator     e.g. "20"
     *
     * Every button press changes the state, and then we redraw the display
     * from the state. The numbers are kept as text while typing, so that
     * "0." or "12.50" look right, and are turned into real numbers only
     * when we calculate.
     *
     * After "=" the result becomes the first number. Pressing an operator
     * keeps calculating with it, pressing a digit starts a new calculation.
     */

    // --- State ---------------------------------------------------------------

    let firstNumber = "";
    let operator = "";
    let secondNumber = "";
    let showingResult = false;   // true right after "=" or "%"

    // --- Elements on the page ------------------------------------------------

    const display = document.getElementById("output");
    const buttons = document.querySelectorAll(".calc-button");

    // --- Helpers -------------------------------------------------------------

    // Show the current state in the display, e.g. "200+20".
    function updateDisplay() {
        display.value = firstNumber + operator + secondNumber;
    }

    // Computers can't store all decimals exactly, so 0.1 + 0.2 gives
    // 0.30000000000000004. Keeping 12 significant digits hides that.
    function round(number) {
        return Number(number.toPrecision(12));
    }

    // Do the actual maths on two numbers.
    function calculate(a, op, b) {
        if (op === "+") return a + b;
        if (op === "-") return a - b;
        if (op === "X") return a * b;
        if (op === "/") return a / b;
    }

    // Percentage maths. "b" is the percentage, e.g. 20 means 20%.
    //   200 X 20%  ->  40    (20% of 200)
    //   200 + 20%  ->  240   (200 plus 20% of 200)
    //   200 - 20%  ->  160   (200 minus 20% of 200)
    //   200 / 20%  ->  1000  (200 divided by 0.2)
    function calculatePercent(a, op, b) {
        const percentOfA = a * (b / 100);

        if (op === "+") return a + percentOfA;
        if (op === "-") return a - percentOfA;
        if (op === "X") return percentOfA;
        if (op === "/") return a / (b / 100);
    }

    // Put a result back into the state so the user can keep calculating with it.
    function showResult(result) {
        if (!isFinite(result)) {
            // Dividing by zero gives Infinity, and broken input like "-" gives
            // NaN ("not a number"). Show an error and start over.
            clearAll();
            display.value = "Error";
            return;
        }

        firstNumber = String(round(result));
        operator = "";
        secondNumber = "";
        showingResult = true;
        updateDisplay();
    }

    // --- What each kind of button does ---------------------------------------

    function clearAll() {
        firstNumber = "";
        operator = "";
        secondNumber = "";
        showingResult = false;
        updateDisplay();
    }

    // Remove the last thing typed: a digit of the second number,
    // then the operator, then digits of the first number.
    function backspace() {
        showingResult = false;
        if (secondNumber !== "") {
            secondNumber = secondNumber.slice(0, -1);
        } else if (operator !== "") {
            operator = "";
        } else {
            firstNumber = firstNumber.slice(0, -1);
        }
        updateDisplay();
    }

    // A digit or the decimal point. Add it to the number we are typing.
    function typeDigit(digit) {
        // A digit right after a result starts a new calculation.
        if (showingResult) {
            clearAll();
        }
        // Before an operator is chosen we are typing the first number.
        let current = operator === "" ? firstNumber : secondNumber;

        // A number can only have one decimal point.
        if (digit === "." && current.includes(".")) {
            return;
        }
        // Write "0.5" instead of ".5".
        if (digit === "." && current === "") {
            digit = "0.";
        }

        current = current + digit;

        if (operator === "") {
            firstNumber = current;
        } else {
            secondNumber = current;
        }
        updateDisplay();
    }

    function chooseOperator(newOperator) {
        // We need a number before an operator.
        if (firstNumber === "") {
            return;
        }
        // "200+20" and then "X": calculate 200+20 first, then continue with X.
        if (secondNumber !== "") {
            pressEquals();
        }
        // If that calculation was an error (like 1/0), everything was cleared.
        if (firstNumber === "") {
            return;
        }
        operator = newOperator;
        showingResult = false;
        updateDisplay();
    }

    function pressEquals() {
        // Nothing to calculate unless we have "number operator number".
        if (secondNumber === "") {
            return;
        }
        const result = calculate(Number(firstNumber), operator, Number(secondNumber));
        showResult(result);
    }

    function pressPercent() {
        if (secondNumber === "") {
            return;
        }
        const result = calculatePercent(Number(firstNumber), operator, Number(secondNumber));
        showResult(result);
    }

    // --- Connect the buttons -------------------------------------------------

    // Every button has a data-val attribute that says what it is.
    function onButtonClick(event) {
        const value = event.currentTarget.dataset.val;

        if (value === "ac") {
            clearAll();
        } else if (value === "<") {
            backspace();
        } else if (value === "=") {
            pressEquals();
        } else if (value === "%") {
            pressPercent();
        } else if (["+", "-", "X", "/"].includes(value)) {
            chooseOperator(value);
        } else {
            typeDigit(value); // 0-9 or "."
        }
    }

    buttons.forEach(function (button) {
        button.addEventListener("click", onButtonClick);
    });
  </script>
</body>
</html>
