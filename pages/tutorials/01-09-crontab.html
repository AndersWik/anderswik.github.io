Crontab checker

We build a crontab checker. There are five boxes, one per time field of a crontab line. Each box gets a red border when what you typed isn't valid, and the help for the field you are editing is shown below.

This is the biggest tutorial in the series. The main lesson is how to keep it small: the five fields follow the same rules with different limits, so we write the rules once and describe each field with a little data.

Open the finished crontab checker

The crontab format

minute  hour  day(month)  month  day(week)  command
30      6     *           *      1-5        /home/me/backup.sh    at 06:30 Monday-Friday
*/15    *     *           *      *          /home/me/check.sh     every 15 minutes

Every field is a list separated by commas, and each part of the list is one of:

*          every value
*/step     every step:th value       */15     -> 0, 15, 30, 45
n          one value                 5
a-b        a range, a <= b           1-5
a-b/step   a range with a step       0-30/10  -> 0, 10, 20, 30
a~b        a random value in a-b     0~30

The limits are:

  • minute: 0-59
  • hour: 0-23
  • day of month: 1-31
  • month: 1-12, or JAN-DEC
  • day of week: 0-7, or SUN-SAT. Both 0 and 7 are Sunday.

Some cron programs also have special symbols for the day fields, like L (last), W (weekday) and # (the n-th weekday). We accept those, but only on their own.

Step 2: Describe the fields with data

const MONTHS = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];
const DAYS = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];

const fields = [
    { id: "minute", min: 0, max: 59 },
    { id: "hour", min: 0, max: 23 },
    {
        id: "day-month", min: 1, max: 31,
        special: /^(\?|L|LW|L-([1-9]|[12][0-9]|30)|([1-9]|[12][0-9]|3[01])W)$/
    },
    { id: "month", min: 1, max: 12, names: MONTHS },
    {
        id: "day-week", min: 0, max: 7, names: DAYS,
        special: /^(\?|L|[0-7]L|[0-7]#[1-5])$/
    }
];

To add a sixth field, for example seconds, you would add one line here. No new code is needed.

The special regular expressions list the extra forms. In a regular expression | means "or", [12][0-9] is 10-29, and \? is a real question mark. Plain ? means "optional".

Step 3: One value

toNumber() turns one value into a number, or NaN if it isn't allowed in this field. Names are looked up in the field's list. MONTHS starts at 1 (JAN) and DAYS at 0 (SUN), the same as the field's min, so the position in the list plus min is the right number.

function toNumber(text, field) {
    if (field.names && field.names.includes(text)) {
        return field.names.indexOf(text) + field.min;
    }
    if (!/^[0-9]+$/.test(text)) {
        return NaN;
    }
    const number = Number(text);
    if (number < field.min || number > field.max) {
        return NaN;
    }
    return number;
}

We check for digits with a regular expression first, because Number() also accepts things like " 5", "1e1" and "0x10", which cron doesn't.

Step 4: One part of the list

split("/") cuts "0-30/10" into ["0-30", "10"]. With destructuring, const [range, step, extra] = ... gives the pieces names. A piece that isn't there is undefined.

function isValidPart(part, field) {
    const [range, step, extra] = part.split("/");

    if (extra !== undefined) {
        return false;                           // more than one "/"
    }
    if (step !== undefined && !/^[1-9][0-9]*$/.test(step)) {
        return false;                           // a step must be 1 or more
    }
    if (range === "*") {
        return true;                            // "*" or "*/step"
    }

    const isRandom = range.includes("~");
    const ends = range.split(isRandom ? "~" : "-");

    if (ends.length === 1) {
        // A single value can't have a step: "5/2" is not allowed.
        return step === undefined && !isNaN(toNumber(ends[0], field));
    }
    if (ends.length !== 2 || (isRandom && step !== undefined)) {
        return false;
    }

    const first = toNumber(ends[0], field);
    const last = toNumber(ends[1], field);

    return first <= last;
}

The last line does two jobs. It checks that the range goes up, so 10-5 is not allowed. And any comparison with NaN is false, so it also fails when either end isn't allowed in this field. Note that NaN === NaN is false too, which is why we need isNaN() to check for it.

Step 5: The whole field

function isValidField(text, field) {
    const value = text.trim().toUpperCase();

    if (field.special && field.special.test(value)) {
        return true;
    }
    return value.split(",").every(function (part) {
        return isValidPart(part, field);
    });
}

toUpperCase() lets you type mon as well as MON. every() is true only if the function is true for every part of the list. Because each part is checked on its own, mixed lists like 1,5-10,*/20 just work. "1,,2" fails, because the empty part between the commas isn't a valid value.

Try it in the console:

isValidField("1,5-10", fields[0])     // true
isValidField("60", fields[0])         // false
isValidField("mon-fri", fields[4])    // true

Step 6: Connect the page

One loop sets up all five boxes:

function showHelp(id) {
    fields.forEach(function (field) {
        document.getElementById(field.id + "-header").hidden = field.id !== id;
        document.getElementById(field.id + "-txt").hidden = field.id !== id;
    });
}

fields.forEach(function (field) {
    const input = document.getElementById(field.id);

    input.addEventListener("focus", function () {
        showHelp(field.id);
    });

    input.addEventListener("input", function () {
        input.classList.toggle("not-valid", !isValidField(input.value, field));
    });
});

field.id !== id is true for every field except the one we want, so all the others are hidden. classList.toggle(name, true) adds the class and toggle(name, false) removes it, so one line replaces an if/else.

Step 7: Test it

These should be valid (no red border):

  • minute: *, */15, 0, 59, 1,5-10, 0-30/10, 0~30, 5-9,10-20/2
  • day of month: 24, 31, L, LW, L-3, 15W, ?
  • month: jan, JAN-MAR, 1-12/3
  • day of week: MON-FRI, 0, 7, 5L, 1#3, mon,wed,fri

These should be red:

  • minute: 60, */0, 5/2, 10-5, 1,,2, -1, 1-2-3
  • day of month: 0, 32, L,5
  • month: 13, MON, DEC-JAN
  • day of week: 8, 1#6, JAN

Click in each box and check that the right help table is shown.

  <script>
    // Crontab checker.
    //
    // A crontab line starts with five time fields, then the command:
    //
    //   minute  hour  day(month)  month  day(week)  command
    //   30      6     *           *      1-5        /home/me/backup.sh
    //
    // Every field is a comma separated list. Each part of the list is one of:
    //
    //   *          every value
    //   */step     every step:th value       */15     -> 0, 15, 30, 45
    //   n          one value                 5
    //   a-b        a range (a <= b)          1-5
    //   a-b/step   a range with a step       0-30/10  -> 0, 10, 20, 30
    //   a~b        a random value in a-b     0~30
    //
    // Month and day(week) can also use names like JAN or MON. Some cron
    // programs add extra symbols (? L W #) to the day fields; we accept them
    // too, but only on their own, not inside a list.
    //
    // When a field is not valid it gets the class "not-valid" (a red border).

    // --- The five fields -----------------------------------------------------

    const MONTHS = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];
    const DAYS = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];

    const fields = [
        { id: "minute", min: 0, max: 59 },
        { id: "hour", min: 0, max: 23 },
        {
            id: "day-month", min: 1, max: 31,
            // ?  any,  L  last day,  LW  last weekday,  L-3  three days before the last,
            // 15W  the weekday closest to the 15th
            special: /^(\?|L|LW|L-([1-9]|[12][0-9]|30)|([1-9]|[12][0-9]|3[01])W)$/
        },
        { id: "month", min: 1, max: 12, names: MONTHS },
        {
            id: "day-week", min: 0, max: 7, names: DAYS,
            // ?  any,  L  last day of the week,  5L  last Friday of the month,
            // 1#3  third Monday of the month
            special: /^(\?|L|[0-7]L|[0-7]#[1-5])$/
        }
    ];

    // --- Checking a field ----------------------------------------------------

    // "5" -> 5, "MON" -> 1, "JAN" -> 1. Returns NaN if it is not allowed here.
    function toNumber(text, field) {
        if (field.names && field.names.includes(text)) {
            // MONTHS starts at 1 (JAN), DAYS starts at 0 (SUN), the same as min.
            return field.names.indexOf(text) + field.min;
        }
        if (!/^[0-9]+$/.test(text)) {
            return NaN;
        }
        const number = Number(text);
        if (number < field.min || number > field.max) {
            return NaN;
        }
        return number;
    }

    // Check one part of a list, e.g. "*/15", "5", "1-5", "0-30/10" or "0~30".
    function isValidPart(part, field) {
        const [range, step, extra] = part.split("/");

        if (extra !== undefined) {
            return false;                           // more than one "/"
        }
        if (step !== undefined && !/^[1-9][0-9]*$/.test(step)) {
            return false;                           // a step must be 1 or more
        }
        if (range === "*") {
            return true;                            // "*" or "*/step"
        }

        const isRandom = range.includes("~");
        const ends = range.split(isRandom ? "~" : "-");

        if (ends.length === 1) {
            // A single value can't have a step: "5/2" is not allowed.
            return step === undefined && !isNaN(toNumber(ends[0], field));
        }
        if (ends.length !== 2 || (isRandom && step !== undefined)) {
            return false;
        }

        const first = toNumber(ends[0], field);
        const last = toNumber(ends[1], field);

        // NaN <= 5 is false, so this also fails when a value is not allowed.
        return first <= last;
    }

    function isValidField(text, field) {
        // Names may be typed in lower case: "mon" is the same as "MON".
        const value = text.trim().toUpperCase();

        if (field.special && field.special.test(value)) {
            return true;
        }
        return value.split(",").every(function (part) {
            return isValidPart(part, field);
        });
    }

    // --- The page ------------------------------------------------------------

    // Show the header and help table for one field, hide the others.
    // The page has e.g. id="minute-header" and id="minute-txt" for "minute".
    function showHelp(id) {
        fields.forEach(function (field) {
            document.getElementById(field.id + "-header").hidden = field.id !== id;
            document.getElementById(field.id + "-txt").hidden = field.id !== id;
        });
    }

    fields.forEach(function (field) {
        const input = document.getElementById(field.id);

        input.addEventListener("focus", function () {
            showHelp(field.id);
        });

        // "input" fires on every change: typing, pasting, deleting...
        input.addEventListener("input", function () {
            // toggle(name, true) adds the class, toggle(name, false) removes it.
            input.classList.toggle("not-valid", !isValidField(input.value, field));
        });
    });
  </script>
