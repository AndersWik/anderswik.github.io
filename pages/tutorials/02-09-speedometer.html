Speedometer

A speedometer for your phone: how fast you go, from the GPS, with the highest speed, the distance and the average speed of the trip. The new ideas are following the phone's position with the Geolocation API, working out a distance on the round Earth, and keeping the screen on.

Open the finished page (on a phone, outdoors: a computer has no GPS)

Read the Compass first. It explains reading a sensor and asking for permission.

The state

let watchId = null;
let wakeLock = null;
let last = null;
let recent = [];
let speed = 0;
let maxSpeed = 0;
let distance = 0;
let movingTime = 0;
let lastFixTime = 0;
let unit = loadUnit();

All speeds are kept in meters a second, the unit the GPS uses, and all distances in meters. They are only turned into km/h or mph when they are shown. Then switching units can never mix them up.

Step 2: Follow the position

navigator.geolocation.watchPosition() asks the browser to call a function every time the phone has a new position, about once a second. The first time, the browser asks the user for permission.

function start() {
    if (!("geolocation" in navigator)) {
        message.textContent = "This browser can't tell your location.";
        return;
    }
    // enableHighAccuracy asks for the GPS, not only Wi-Fi and phone masts.
    // maximumAge: 0 means always a new position, never an old saved one.
    watchId = navigator.geolocation.watchPosition(onPosition, onError, {
        enableHighAccuracy: true,
        maximumAge: 0,
        timeout: 10000
    });
    last = null;   // don't count the distance from where we stopped last time
    recent = [];
    lastFixTime = 0;
    keepScreenOn();
    startButton.textContent = "Stop";
    startButton.className = "btn btn-warning";
    showGps("Looking for GPS...", "poor");
    message.textContent = "Waiting for the first position.";
}
  • enableHighAccuracy: true asks for the GPS. Without it, the phone may only use Wi-Fi and phone masts, which can be hundreds of meters off and has no speed.
  • maximumAge: 0 means a new position every time, never one the phone saved earlier.
  • watchPosition returns an id, like setInterval. clearWatch(watchId) stops it again, in stop().
  • The Geolocation API only works on pages served over https:// (or localhost).

If something goes wrong, the browser calls onError instead. The worst case is that the user said no: then we can only tell them how to change it.

function onError(error) {
    if (error.code === error.PERMISSION_DENIED) {
        message.textContent = "The page may not use your location. Allow it in the browser's settings.";
        stop();
    } else if (error.code === error.POSITION_UNAVAILABLE) {
        showGps("No GPS signal", "none");
    }
    // TIMEOUT: watchPosition keeps trying by itself.
}

Step 3: The speed

Each position has coords.latitude and coords.longitude, and on most phones also coords.speed in meters a second. The GPS measures the speed very well, from the tiny change in pitch of the signals from the satellites (the Doppler effect, the same thing you hear when an ambulance drives past). It is often more exact than the car's own speedometer, which shows a few percent too much on purpose.

function onPosition(position) {
    const coords = position.coords;
    const time = position.timestamp;
    lastFixTime = Date.now();

    const accuracy = coords.accuracy;   // meters, about: the real place is within this
    // \u00B1 is the ± sign.
    showGps("GPS \u00B1" + Math.round(accuracy) + " m", accuracy <= GOOD_ACCURACY ? "good" : "poor");
    message.textContent = "Measuring.";

    // How far and how long since the last position.
    let step = 0;
    let seconds = 0;
    if (last !== null) {
        step = distanceBetween(last.latitude, last.longitude, coords.latitude, coords.longitude);
        seconds = (time - last.time) / 1000;
    }

    // Keep the positions of the last few seconds. The oldest one stays until
    // the next one is old enough, so there is always a few seconds' worth.
    recent.push({ latitude: coords.latitude, longitude: coords.longitude, time: time });
    while (recent.length > 1 && time - recent[1].time >= SPEED_WINDOW) {
        recent.shift();
    }

    if (coords.speed !== null && !isNaN(coords.speed)) {
        // The phone knows the speed: use it.
        speed = coords.speed;
    } else {
        // Work it out ourselves: how far from the oldest of the recent
        // positions, divided by the time. Over a few seconds, the GPS
        // wobbling a meter or two back and forth hardly counts.
        const oldest = recent[0];
        const windowSeconds = (time - oldest.time) / 1000;
        speed = windowSeconds > 0
            ? distanceBetween(oldest.latitude, oldest.longitude, coords.latitude, coords.longitude) / windowSeconds
            : 0;
    }

    // Standing still, GPS still wobbles a few meters. Call that 0.
    if (speed < STILL_SPEED) {
        speed = 0;
    }

    // Only count distance and max from positions we can trust, and only
    // when moving, or the wobble adds up to meters while standing still.
    if (speed > 0 && accuracy <= POOR_ACCURACY) {
        distance += step;
        movingTime += seconds;
        maxSpeed = Math.max(maxSpeed, speed);
    }

    last = { latitude: coords.latitude, longitude: coords.longitude, time: time };
    showNumbers();
}
  • If the phone gives no speed, coords.speed is null, and we work it out: how far from the oldest of the positions of the last 4 seconds, divided by the time. Measuring over a few seconds matters. A GPS position wobbles a meter or two back and forth even when you stand still. Over one second, that looks like walking speed; over four seconds, it hardly counts.
  • recent.shift() removes the first position in the list. The loop keeps the oldest one until the one after it is also 4 seconds old, so there is always about 4 seconds of positions.
  • Anything slower than STILL_SPEED, about 3 km/h, is shown as 0, so the speedometer doesn't show 2 km/h at a red light.
  • Distance and max speed only count when we are moving, and when coords.accuracy is good enough. Otherwise the wobble adds up to meters while you stand still, and one bad position could set a max speed of 200 km/h.

Step 4: Distance on a round Earth

Latitude and longitude are angles, not meters, and the lines of longitude get closer together towards the poles. So Pythagoras on the numbers doesn't work. The haversine formula gives the distance along the Earth's surface between two points:

function toRadians(degrees) {
    return degrees * Math.PI / 180;
}

function distanceBetween(lat1, lon1, lat2, lon2) {
    const dLat = toRadians(lat2 - lat1);
    const dLon = toRadians(lon2 - lon1);
    const a = Math.sin(dLat / 2) ** 2 +
        Math.cos(toRadians(lat1)) * Math.cos(toRadians(lat2)) * Math.sin(dLon / 2) ** 2;
    return 2 * EARTH_RADIUS * Math.asin(Math.sqrt(a));
}

You don't need to follow the math to use it. It works out the angle between the two points as seen from the middle of the Earth, and multiplies by the Earth's radius, 6371 km. x ** 2 is x squared, the same as x * x.

Between two positions a second apart, the Earth is almost flat, so a simpler formula would do. But the haversine formula is correct for any distance, also from Stockholm to New York.

Step 5: Show it, in km/h or mph

// meters a second -> the unit on the screen, and meters -> km or miles
const UNITS = {
    kmh: { speed: 3.6, speedName: "km/h", distance: 1 / 1000, distanceName: "km", other: "Show mph" },
    mph: { speed: 2.23694, speedName: "mph", distance: 1 / 1609.344, distanceName: "mi", other: "Show km/h" }
};

function showNumbers() {
    const u = UNITS[unit];
    speedText.textContent = Math.round(speed * u.speed);
    unitText.textContent = u.speedName;
    maxText.textContent = Math.round(maxSpeed * u.speed);
    // 2 decimals, but only 1 from 100 km on, so it fits on a phone.
    const shown = distance * u.distance;
    distanceText.textContent = shown.toFixed(shown < 100 ? 2 : 1) + " " + u.distanceName;
    const average = movingTime > 0 ? distance / movingTime : 0;
    averageText.textContent = Math.round(average * u.speed);
    unitButton.textContent = u.other;
}

Each unit is an object with everything that changes: how to convert a speed (1 m/s is 3.6 km/h or 2.237 mph), how to convert a distance, and the names. UNITS[unit] picks the right one, so showNumbers() needs no if at all. The average is the distance divided by the time spent moving, so stops at red lights don't pull it down.

The chosen unit is saved with localStorage, so a user in the US only has to choose mph once.

Step 6: No signal

In a tunnel the GPS loses the satellites, and positions just stop coming. No event tells us that, so we check once a second when the last one came:

function checkSignal() {
    if (watchId !== null && lastFixTime > 0 && Date.now() - lastFixTime > SIGNAL_TIMEOUT) {
        showGps("No GPS signal", "none");
        speed = 0;
        showNumbers();
    }
}

setInterval(checkSignal, 1000);

Step 7: Keep the screen on

When the screen turns off, the browser stops sending positions, to save battery. The Screen Wake Lock API asks to keep the screen on:

async function keepScreenOn() {
    if (!("wakeLock" in navigator)) {
        return;
    }
    try {
        wakeLock = await navigator.wakeLock.request("screen");
    } catch (error) {
        wakeLock = null;
    }
}

// The lock is let go when the page is hidden, so ask again when it comes back.
document.addEventListener("visibilitychange", function () {
    if (document.visibilityState === "visible" && watchId !== null) {
        keepScreenOn();
    }
});
  • navigator.wakeLock.request("screen") returns a promise, so keepScreenOn is async and awaits it, as in the Bubble level.
  • Not every browser has it, so we check for it first, and a try catches a no.
  • The browser lets go of the lock when the page is hidden, for example when you switch apps. The visibilitychange event tells us when the page comes back, and we ask again.
  • stop() calls wakeLock.release(), so the screen can turn off as usual again.

Step 8: Test it

  • Put the page on a server with https://, and open it on a phone, outdoors. Indoors, the GPS often can't see the satellites.
  • Press Start and say no to the location. The message should say how to allow it.
  • Stand still for a minute. The speed should stay at 0, and the distance shouldn't grow.
  • Walk, cycle or ride as a passenger in a car, and compare with the car's speedometer. The app should show a little less: the car's shows too much.
  • Switch to mph and reload. It should still show mph.
  • On a computer, Firefox and Chrome can fake a position, but no speed. That is a good way to test the haversine fallback.

  <script>
    /*
     * Speedometer: how fast you go, from the phone's GPS.
     *
     * navigator.geolocation.watchPosition() calls us with a new position about
     * once a second. Most phones also tell the speed (coords.speed, in meters a
     * second), which the GPS measures very well. If it doesn't, we work the
     * speed out ourselves: the distance between the last two positions divided
     * by the time between them.
     *
     * State:
     *   watchId       the running watchPosition (null when stopped)
     *   wakeLock      keeps the screen on (null when we don't have one)
     *   last          the last position: latitude, longitude and time
     *   recent        the positions of the last few seconds, for working out the speed
     *   speed         the speed now, in meters a second
     *   maxSpeed      the highest speed of the trip, in meters a second
     *   distance      how far the trip has gone, in meters
     *   movingTime    how long we have been moving, in seconds, for the average
     *   lastFixTime   when the last position came, to notice a lost signal
     *   unit          "kmh" or "mph"
     */

    // --- Settings --------------------------------------------------------------

    const SPEED_WINDOW = 4000;  // milliseconds: work out the speed over this long
    const STILL_SPEED = 0.8;     // m/s, about 3 km/h: slower than this counts as standing still
    const GOOD_ACCURACY = 20;    // meters: a position better than this can be trusted
    const POOR_ACCURACY = 50;    // meters: worse than this, don't count distance or max
    const SIGNAL_TIMEOUT = 5000; // milliseconds without a position before we say "no signal"
    const EARTH_RADIUS = 6371000; // meters
    const UNIT_KEY = "speedometer-unit";

    // meters a second -> the unit on the screen, and meters -> km or miles
    const UNITS = {
        kmh: { speed: 3.6, speedName: "km/h", distance: 1 / 1000, distanceName: "km", other: "Show mph" },
        mph: { speed: 2.23694, speedName: "mph", distance: 1 / 1609.344, distanceName: "mi", other: "Show km/h" }
    };

    // --- State -----------------------------------------------------------------

    let watchId = null;
    let wakeLock = null;
    let last = null;
    let recent = [];
    let speed = 0;
    let maxSpeed = 0;
    let distance = 0;
    let movingTime = 0;
    let lastFixTime = 0;
    let unit = loadUnit();

    // --- Elements on the page --------------------------------------------------

    const speedText = document.getElementById("speed");
    const unitText = document.getElementById("unit");
    const gpsText = document.getElementById("gps");
    const maxText = document.getElementById("max");
    const distanceText = document.getElementById("distance");
    const averageText = document.getElementById("average");
    const message = document.getElementById("message");
    const startButton = document.getElementById("button-start");
    const unitButton = document.getElementById("button-unit");
    const resetButton = document.getElementById("button-reset");

    // --- Helpers ---------------------------------------------------------------

    function toRadians(degrees) {
        return degrees * Math.PI / 180;
    }

    // The distance in meters between two places on the Earth, from their
    // latitude and longitude: the haversine formula. The Earth is round, so we
    // can't use Pythagoras on degrees.
    function distanceBetween(lat1, lon1, lat2, lon2) {
        const dLat = toRadians(lat2 - lat1);
        const dLon = toRadians(lon2 - lon1);
        const a = Math.sin(dLat / 2) ** 2 +
            Math.cos(toRadians(lat1)) * Math.cos(toRadians(lat2)) * Math.sin(dLon / 2) ** 2;
        return 2 * EARTH_RADIUS * Math.asin(Math.sqrt(a));
    }

    function loadUnit() {
        try {
            return localStorage.getItem(UNIT_KEY) === "mph" ? "mph" : "kmh";
        } catch (error) {
            return "kmh";
        }
    }

    function saveUnit() {
        try {
            localStorage.setItem(UNIT_KEY, unit);
        } catch (error) {
            // Not saved, but the app still works.
        }
    }

    // --- Drawing ---------------------------------------------------------------

    function showNumbers() {
        const u = UNITS[unit];
        speedText.textContent = Math.round(speed * u.speed);
        unitText.textContent = u.speedName;
        maxText.textContent = Math.round(maxSpeed * u.speed);
        // 2 decimals, but only 1 from 100 km on, so it fits on a phone.
        const shown = distance * u.distance;
        distanceText.textContent = shown.toFixed(shown < 100 ? 2 : 1) + " " + u.distanceName;
        const average = movingTime > 0 ? distance / movingTime : 0;
        averageText.textContent = Math.round(average * u.speed);
        unitButton.textContent = u.other;
    }

    // How much to trust the GPS: the class colors the text.
    function showGps(text, quality) {
        gpsText.textContent = text;
        gpsText.className = "gps " + quality;
    }

    // --- The GPS ---------------------------------------------------------------

    function onPosition(position) {
        const coords = position.coords;
        const time = position.timestamp;
        lastFixTime = Date.now();

        const accuracy = coords.accuracy;   // meters, about: the real place is within this
        // \u00B1 is the ± sign.
        showGps("GPS \u00B1" + Math.round(accuracy) + " m", accuracy <= GOOD_ACCURACY ? "good" : "poor");
        message.textContent = "Measuring.";

        // How far and how long since the last position.
        let step = 0;
        let seconds = 0;
        if (last !== null) {
            step = distanceBetween(last.latitude, last.longitude, coords.latitude, coords.longitude);
            seconds = (time - last.time) / 1000;
        }

        // Keep the positions of the last few seconds. The oldest one stays until
        // the next one is old enough, so there is always a few seconds' worth.
        recent.push({ latitude: coords.latitude, longitude: coords.longitude, time: time });
        while (recent.length > 1 && time - recent[1].time >= SPEED_WINDOW) {
            recent.shift();
        }

        if (coords.speed !== null && !isNaN(coords.speed)) {
            // The phone knows the speed: use it.
            speed = coords.speed;
        } else {
            // Work it out ourselves: how far from the oldest of the recent
            // positions, divided by the time. Over a few seconds, the GPS
            // wobbling a meter or two back and forth hardly counts.
            const oldest = recent[0];
            const windowSeconds = (time - oldest.time) / 1000;
            speed = windowSeconds > 0
                ? distanceBetween(oldest.latitude, oldest.longitude, coords.latitude, coords.longitude) / windowSeconds
                : 0;
        }

        // Standing still, GPS still wobbles a few meters. Call that 0.
        if (speed < STILL_SPEED) {
            speed = 0;
        }

        // Only count distance and max from positions we can trust, and only
        // when moving, or the wobble adds up to meters while standing still.
        if (speed > 0 && accuracy <= POOR_ACCURACY) {
            distance += step;
            movingTime += seconds;
            maxSpeed = Math.max(maxSpeed, speed);
        }

        last = { latitude: coords.latitude, longitude: coords.longitude, time: time };
        showNumbers();
    }

    function onError(error) {
        if (error.code === error.PERMISSION_DENIED) {
            message.textContent = "The page may not use your location. Allow it in the browser's settings.";
            stop();
        } else if (error.code === error.POSITION_UNAVAILABLE) {
            showGps("No GPS signal", "none");
        }
        // TIMEOUT: watchPosition keeps trying by itself.
    }

    // No position for a while: in a tunnel, or the GPS lost the satellites.
    function checkSignal() {
        if (watchId !== null && lastFixTime > 0 && Date.now() - lastFixTime > SIGNAL_TIMEOUT) {
            showGps("No GPS signal", "none");
            speed = 0;
            showNumbers();
        }
    }

    // --- Keeping the screen on -------------------------------------------------

    // The browser stops sending positions when the screen turns off, so we ask
    // to keep it on. Not every browser can, so we try.
    async function keepScreenOn() {
        if (!("wakeLock" in navigator)) {
            return;
        }
        try {
            wakeLock = await navigator.wakeLock.request("screen");
        } catch (error) {
            wakeLock = null;
        }
    }

    // The lock is let go when the page is hidden, so ask again when it comes back.
    document.addEventListener("visibilitychange", function () {
        if (document.visibilityState === "visible" && watchId !== null) {
            keepScreenOn();
        }
    });

    // --- Start and stop --------------------------------------------------------

    function start() {
        if (!("geolocation" in navigator)) {
            message.textContent = "This browser can't tell your location.";
            return;
        }
        // enableHighAccuracy asks for the GPS, not only Wi-Fi and phone masts.
        // maximumAge: 0 means always a new position, never an old saved one.
        watchId = navigator.geolocation.watchPosition(onPosition, onError, {
            enableHighAccuracy: true,
            maximumAge: 0,
            timeout: 10000
        });
        last = null;   // don't count the distance from where we stopped last time
        recent = [];
        lastFixTime = 0;
        keepScreenOn();
        startButton.textContent = "Stop";
        startButton.className = "btn btn-warning";
        showGps("Looking for GPS...", "poor");
        message.textContent = "Waiting for the first position.";
    }

    function stop() {
        if (watchId !== null) {
            navigator.geolocation.clearWatch(watchId);
        }
        watchId = null;
        if (wakeLock !== null) {
            wakeLock.release();
            wakeLock = null;
        }
        speed = 0;
        startButton.textContent = "Start";
        startButton.className = "btn btn-success";
        showGps("GPS off", "");
        showNumbers();
    }

    function reset() {
        maxSpeed = 0;
        distance = 0;
        movingTime = 0;
        showNumbers();
    }

    function toggleUnit() {
        unit = unit === "kmh" ? "mph" : "kmh";
        saveUnit();
        showNumbers();
    }

    // --- Connect the buttons ---------------------------------------------------

    startButton.addEventListener("click", function () {
        if (watchId === null) {
            start();
        } else {
            stop();
        }
    });
    unitButton.addEventListener("click", toggleUnit);
    resetButton.addEventListener("click", reset);

    setInterval(checkSignal, 1000);
    showNumbers();
  </script>
</body>
