Base64 encoder

We build a Base64 encoder. The page looks almost exactly like the URL encoder, but the JavaScript has to deal with a classic problem: text and bytes are not the same thing.

Open the finished Base64 encoder

What Base64 is

Base64 writes any data, text or images, using only 64 safe characters: A-Z, a-z, 0-9, + and /, plus = as padding at the end. It is used where only plain text is allowed, like email attachments, data URLs and tokens in JSON. Every 3 bytes become 4 characters, so the result is about a third longer.

Base64 is not encryption. Anyone can decode it.

The problem with btoa and atob

The browser has btoa() (binary to ASCII, which encodes) and atob() (which decodes). They were made long ago and only accept characters with codes 0-255, one character per byte:

btoa("hello")   // "aGVsbG8="
btoa("åäö")     // works, but gives the wrong answer for UTF-8
btoa("😀")      // error: "The string contains characters outside of the Latin1 range"

Text on the web is UTF-8, where å is two bytes and 😀 is four. So we first turn the text into UTF-8 bytes, and then give btoa() one character per byte.

Step 2: Text to bytes

TextEncoder turns text into UTF-8 bytes. The result is a Uint8Array, a list of numbers from 0 to 255. Try it in the console:

new TextEncoder().encode("åä")   // Uint8Array [195, 165, 195, 164]

String.fromCharCode(195) gives the character with code 195. We do that for every byte and join the characters into a string, which btoa() accepts:

// "åäö" -> "w6XDpMO2"
function textToBase64(text) {
    const bytes = new TextEncoder().encode(text);

    let binary = "";
    bytes.forEach(function (byte) {
        binary += String.fromCharCode(byte);
    });

    return btoa(binary);
}

Step 3: Bytes back to text

Decoding is the same steps in reverse. atob() gives one character per byte. charCodeAt(0) gives the number of a character, and Uint8Array.from() builds the byte list. TextDecoder turns UTF-8 bytes back into text:

// "w6XDpMO2" -> "åäö"
function base64ToText(base64) {
    const binary = atob(base64.trim());

    const bytes = Uint8Array.from(binary, function (character) {
        return character.charCodeAt(0);
    });

    return new TextDecoder().decode(bytes);
}

trim() removes spaces and new lines at the start and end, which are easy to get when you copy and paste.

Step 4: The buttons

atob() throws an error if the text is not valid Base64, so decoding goes in a try:

function encode() {
    base64Text.value = textToBase64(plainText.value);
}

function decode() {
    try {
        plainText.value = base64ToText(base64Text.value);
    } catch (error) {
        plainText.value = "That is not valid Base64.";
    }
}

encodeButton.addEventListener("click", encode);
decodeButton.addEventListener("click", decode);

<body>
  <h1>Base64 encoder</h1>

  <label for="decoded-text">Text</label>
  <textarea id="decoded-text"></textarea>

  <label for="encoded-text">Base64 encoded text</label>
  <textarea id="encoded-text"></textarea>

  <button type="button" id="encode-btn">Encode</button>
  <button type="button" id="decode-btn">Decode</button>

  <script>
    /*
     * Base64 encode / decode.
     *
     * Base64 turns any data into plain letters, digits, "+" and "/", so it can
     * be sent where only text is allowed (emails, URLs, JSON...).
     *
     * The browser has two built-in functions for this: btoa() encodes and
     * atob() decodes. They only understand simple English characters, so
     * text like "åäö" or emojis must first be turned into bytes (UTF-8).
     */

    // --- Elements on the page ------------------------------------------------

    const plainText = document.getElementById("decoded-text");
    const base64Text = document.getElementById("encoded-text");
    const encodeButton = document.getElementById("encode-btn");
    const decodeButton = document.getElementById("decode-btn");

    // --- Helpers -------------------------------------------------------------

    // "åäö" -> "w6XDpMO2"
    function textToBase64(text) {
        // Turn the text into UTF-8 bytes (numbers 0-255).
        const bytes = new TextEncoder().encode(text);

        // btoa() wants one character per byte, so build such a string.
        let binary = "";
        bytes.forEach(function (byte) {
            binary += String.fromCharCode(byte);
        });

        return btoa(binary);
    }

    // "w6XDpMO2" -> "åäö"
    function base64ToText(base64) {
        // atob() gives one character per byte...
        const binary = atob(base64.trim());

        // ...turn them back into bytes, and the bytes back into text.
        const bytes = Uint8Array.from(binary, function (character) {
            return character.charCodeAt(0);
        });

        return new TextDecoder().decode(bytes);
    }

    // --- What the buttons do -------------------------------------------------

    function encode() {
        base64Text.value = textToBase64(plainText.value);
    }

    function decode() {
        // atob() throws an error if the text is not valid Base64.
        try {
            plainText.value = base64ToText(base64Text.value);
        } catch (error) {
            plainText.value = "That is not valid Base64.";
        }
    }

    // --- Connect the buttons -------------------------------------------------

    encodeButton.addEventListener("click", encode);
    decodeButton.addEventListener("click", decode);
  </script>
</body>
</html>
