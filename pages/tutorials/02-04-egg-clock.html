Egg clock

An egg clock: set the time with buttons, press Start, and it beeps when it reaches zero. The new ideas are keeping a timer as state, so it can't run twice, and making a sound with the Web Audio API, without a sound file.

Open the finished page

Read the Countdown first. It explains setInterval and clearInterval.

The state

The clock has to remember three things:

let secondsLeft = 0;  // how much time is left, in seconds
let timer = null;     // the running countdown (null when stopped)
let beeper = null;    // the repeating beep (null when quiet)
let audio = null;     // for making sound, see step 5

const MAX_SECONDS = 60 * 60; // one hour

timer and beeper hold the ids that setInterval returns, and null when nothing is running. Then we can always tell whether the clock is running, and stop it.

Step 2: Show the time

function twoDigits(number) {
    return String(number).padStart(2, "0");
}

// 125 seconds -> "02" minutes and "05" seconds
function showTime() {
    minutesText.textContent = twoDigits(Math.floor(secondsLeft / 60));
    secondsText.textContent = twoDigits(secondsLeft % 60);
}

Step 3: Set the time

One function handles all four buttons. A positive number adds time and a negative number removes it. Math.min and Math.max keep the time between 0 and one hour:

function changeTime(seconds) {
    secondsLeft = Math.min(MAX_SECONDS, Math.max(0, secondsLeft + seconds));
    showTime();
}

addMinuteButton.addEventListener("click", function () { changeTime(60); });
removeMinuteButton.addEventListener("click", function () { changeTime(-60); });
addSecondButton.addEventListener("click", function () { changeTime(1); });
removeSecondButton.addEventListener("click", function () { changeTime(-1); });

addEventListener needs a function to call later. changeTime(60) would run right away, so we wrap it in a small function without a name: function () { changeTime(60); }.

Step 4: Start, tick and stop

function start() {
    if (audio === null) {
        audio = new AudioContext();
    }
    if (timer !== null || secondsLeft === 0) {
        return;
    }
    clearInterval(beeper);
    beeper = null;
    timer = setInterval(tick, 1000);
}

The check timer !== null matters. Without it, pressing Start twice would start two intervals, the clock would run at double speed, and Stop could only stop one of them. If it is still beeping from last time, we stop that too.

function tick() {
    if (secondsLeft > 0) {
        secondsLeft = secondsLeft - 1;
        showTime();
    }
    if (secondsLeft === 0) {
        clearInterval(timer);
        timer = null;
        beep();
        beeper = setInterval(beep, 1000);
    }
}

Why not just subtract 1 and check for 0? Because the user can press "- 1 min" while the clock is running and set the time to 0 between two ticks. If we always subtracted, the next tick would give -1, which never equals 0, and the clock would count down into negative numbers forever. So we only subtract when there is time left, and the zero check catches both cases.

function stop() {
    clearInterval(timer);
    clearInterval(beeper);
    timer = null;
    beeper = null;
}

Stop works as both pause and silence. clearInterval(null) does nothing, so we don't need to check first.

Step 5: A beep without a sound file

The Web Audio API can make sound from code. An oscillator makes a steady tone. We connect it to the speakers, start it, and tell it to stop after 0.3 seconds:

function beep() {
    const tone = audio.createOscillator();
    tone.frequency.value = 880;              // pitch in hertz, a high A
    tone.connect(audio.destination);         // to the speakers
    tone.start();
    tone.stop(audio.currentTime + 0.3);      // 0.3 seconds long
}

An oscillator can only be played once, so beep() makes a new one every time.

Browsers don't let a page make sound until the user has clicked something, so pages can't start shouting by themselves. That is why audio = new AudioContext() is in start(), which runs on a click, and not at the top of the script.

Step 6: Test it

  • Set 5 seconds and press Start. It should count down and beep once a second at zero. Stop silences it.
  • Press Start twice quickly. The clock should still count one second per second.
  • Set 1 minute, press Start, then press "- 1 min" right away. It should stop at 00:00 and beep, not show negative numbers.
  • While it beeps, add 5 seconds and press Start. The beep stops and it counts down again.
  • Press "- 1 min" at 00:30. It goes to 00:00.

  <script>
    /*
     * Egg clock: count down from a chosen time and beep when it reaches zero.
     *
     * State:
     *   secondsLeft  how much time is left, in seconds
     *   timer        the running countdown (null when stopped)
     *   beeper       the repeating beep (null when quiet)
     *
     * setInterval(fn, 1000) calls fn once every second until clearInterval()
     * stops it. It returns an id that we keep so we can stop it later.
     */

    // --- State ---------------------------------------------------------------

    let secondsLeft = 0;
    let timer = null;
    let beeper = null;
    let audio = null;   // made on the first Start, see beep()

    const MAX_SECONDS = 60 * 60; // one hour

    // --- Elements on the page ------------------------------------------------

    const minutesText = document.getElementById("output-minutes");
    const secondsText = document.getElementById("output-seconds");
    const startButton = document.getElementById("button-start");
    const stopButton = document.getElementById("button-stop");
    const addMinuteButton = document.getElementById("increase-minutes");
    const removeMinuteButton = document.getElementById("decrease-minutes");
    const addSecondButton = document.getElementById("increase-seconds");
    const removeSecondButton = document.getElementById("decrease-seconds");

    // --- Helpers -------------------------------------------------------------

    // 7 -> "07"
    function twoDigits(number) {
        return String(number).padStart(2, "0");
    }

    // 125 seconds -> "02" minutes and "05" seconds
    function showTime() {
        minutesText.textContent = twoDigits(Math.floor(secondsLeft / 60));
        secondsText.textContent = twoDigits(secondsLeft % 60);
    }

    // Make a short beep with the Web Audio API, so we need no sound file.
    // An oscillator makes a tone; 880 is the pitch in hertz (a high A).
    function beep() {
        const tone = audio.createOscillator();
        tone.frequency.value = 880;
        tone.connect(audio.destination);       // to the speakers
        tone.start();
        tone.stop(audio.currentTime + 0.3);    // 0.3 seconds long
    }

    // --- Actions -------------------------------------------------------------

    // Add (positive) or remove (negative) time, but stay within 0 and one hour.
    function changeTime(seconds) {
        secondsLeft = Math.min(MAX_SECONDS, Math.max(0, secondsLeft + seconds));
        showTime();
    }

    // Called once every second while the clock runs.
    function tick() {
        if (secondsLeft > 0) {
            secondsLeft = secondsLeft - 1;
            showTime();
        }
        // Also true if the time was set to 0 with the buttons while running.
        if (secondsLeft === 0) {
            clearInterval(timer);
            timer = null;
            beep();
            beeper = setInterval(beep, 1000);
        }
    }

    function start() {
        // Browsers only allow sound after the user has clicked something,
        // so the audio is made here and not when the page loads.
        if (audio === null) {
            audio = new AudioContext();
        }
        // Don't start twice (the clock would run at double speed),
        // and there is nothing to count down from 0.
        if (timer !== null || secondsLeft === 0) {
            return;
        }
        // Stop a beep that is still going from the last time.
        clearInterval(beeper);
        beeper = null;
        timer = setInterval(tick, 1000);
    }

    // Pause the countdown and stop any beeping.
    function stop() {
        clearInterval(timer);
        clearInterval(beeper);
        timer = null;
        beeper = null;
    }

    // --- Connect the buttons -------------------------------------------------

    startButton.addEventListener("click", start);
    stopButton.addEventListener("click", stop);

    addMinuteButton.addEventListener("click", function () { changeTime(60); });
    removeMinuteButton.addEventListener("click", function () { changeTime(-60); });
    addSecondButton.addEventListener("click", function () { changeTime(1); });
    removeSecondButton.addEventListener("click", function () { changeTime(-1); });
  </script>
</body>
</html>

Try this

  • Add preset buttons, like "Soft boiled (6 min)" and "Hard boiled (10 min)".
