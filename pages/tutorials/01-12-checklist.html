Checklist

We build a checklist: add items, tick them off, delete them, and clear the ones that are done. The list is saved in the browser, so it is still there when you come back.

The new ideas are saving data with localStorage, and building the list from the state with createElement.

Open the finished page

The state

The whole checklist is one list of objects:

let items = [
    { text: "Buy milk", done: true },
    { text: "Water the plants", done: false }
];

Like the calculator, every action changes the state and then one function, render(), draws the page from it. Here every action also saves the state.

Step 1: The page

Create checklist.html from the skeleton.

<h1>Checklist</h1>

<div class="card">
  <p class="label">Checklist</p>

  <button type="button" id="add">+ Add an item</button>

  <form id="form" hidden>
    <input type="text" id="new-item" maxlength="200" placeholder="What needs doing?" aria-label="New item" autocomplete="off">
  </form>

  <ul id="items"></ul>

  <p id="empty">Nothing here yet</p>

  <div id="footer" hidden>
    <span id="count"></span>
    <button type="button" id="clear">Clear done</button>
  </div>
</div>

The <ul> starts empty. JavaScript fills it. The hidden attribute hides an element, and in JavaScript element.hidden = true or false hides or shows it. We add [hidden] { display: none !important; } to the CSS, so that a display: flex rule can't make a hidden element show up again.

The text field is inside a <form>. Then pressing Enter in it submits the form, and we don't have to listen for the Enter key ourselves.

Step 2: Save and load

localStorage keeps text in the browser, also after a reload or a restart. It belongs to the site, so other sites can't read it. It can only store text, so we turn the list into JSON text first:

const STORAGE_KEY = "checklist-items";

function save() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
}

function load() {
    try {
        const saved = JSON.parse(localStorage.getItem(STORAGE_KEY));
        return Array.isArray(saved) ? saved : [];
    } catch (error) {
        return [];
    }
}

let items = load();

JSON.stringify turns the list into text like [{"text":"Buy milk","done":true}], and JSON.parse turns it back. The first time, nothing is saved and getItem gives null. JSON.parse(null) is null, which is not an array, so we start with []. If the saved text is broken, JSON.parse throws an error, and try / catch gives an empty list then too, instead of a page that doesn't work.

Step 3: Build one item

Each item becomes an <li> with a checkbox, the text and a delete button. document.createElement makes a new element, and append puts elements inside another:

function makeItem(item, index) {
    const li = document.createElement("li");
    if (item.done) {
        li.classList.add("done");
    }

    const box = document.createElement("input");
    box.type = "checkbox";
    box.checked = item.done;
    box.addEventListener("change", function () { toggle(index); });

    const text = document.createElement("span");
    text.textContent = item.text;

    const label = document.createElement("label");
    label.append(box, text);

    const deleteButton = document.createElement("button");
    deleteButton.type = "button";
    deleteButton.className = "delete";
    deleteButton.textContent = "×";
    deleteButton.setAttribute("aria-label", "Delete " + item.text);
    deleteButton.addEventListener("click", function () { remove(index); });

    li.append(label, deleteButton);
    return li;
}

The checkbox is inside the <label>, so a click on the text ticks the box too.

Why textContent and not innerHTML? It would be shorter to write li.innerHTML = "<span>" + item.text + "</span>". But then a user who types <b>today</b> gets bold text, and worse, <img src=x onerror=alert(1)> would run code. textContent always shows the text exactly as typed. Use it whenever you show text that a user wrote.

Step 4: Draw the list

function render() {
    list.textContent = "";   // remove the old items
    items.forEach(function (item, index) {
        list.append(makeItem(item, index));
    });

    const done = items.filter(function (item) { return item.done; }).length;
    count.textContent = done + " of " + items.length + " done";
    clearButton.disabled = done === 0;
    footer.hidden = items.length === 0;
    showEmpty();
}

function showEmpty() {
    empty.hidden = items.length > 0 || !form.hidden;
}

Setting textContent to "" removes everything inside the list. Then we build it again from the state. Building the whole list every time is simple and fast enough for a list of a few hundred items.

filter makes a new list with only the items where the function returns true, so .length of it is the number of items that are done.

Step 5: The actions

function add(text) {
    items.push({ text: text, done: false });
    save();
    render();
}

function toggle(index) {
    items[index].done = !items[index].done;
    save();
    render();
}

function remove(index) {
    items.splice(index, 1);   // remove 1 item at index
    save();
    render();
}

function clearDone() {
    items = items.filter(function (item) { return !item.done; });
    save();
    render();
}

Every action is the same three steps: change the state, save, draw. ! turns true into false and back.

Step 6: The text field

"Add an item" hides itself and shows the text field. Enter adds the item and keeps the field open for the next one. Escape closes it:

function openForm() {
    addButton.hidden = true;
    form.hidden = false;
    input.focus();
    showEmpty();
}

function closeForm() {
    input.value = "";
    form.hidden = true;
    addButton.hidden = false;
    showEmpty();
}

addButton.addEventListener("click", openForm);

form.addEventListener("submit", function (event) {
    event.preventDefault();   // don't reload the page
    const text = input.value.trim();
    if (text !== "") {
        add(text);
        input.value = "";
    }
});

input.addEventListener("keydown", function (event) {
    if (event.key === "Escape") {
        closeForm();
    }
});

input.addEventListener("blur", function () {
    if (input.value.trim() === "") {
        closeForm();
    }
});

A form normally sends its data to a server and loads a new page. event.preventDefault() stops that. trim() removes spaces at the start and end, so an item of only spaces is not added.

blur happens when the field loses focus, for example when you click somewhere else. Then an empty field closes by itself.

A bug to avoid. Notice that openForm and closeForm call showEmpty() and not render(). Say the field is open and empty, and you click a checkbox. The browser first sends blur to the field, and only then the click to the checkbox. If closeForm called render(), the list would be rebuilt in between, the checkbox you pressed would be replaced by a new one, and the click would be lost. So only change what you need to.

Step 7: Test it

  • Add three items with Enter. Reload the page. They are still there.
  • Tick one. It is crossed out and the footer says 1 of 3 done. Press "Clear done" and it is gone.
  • Add an item called <b>bold</b>. It shows the tags as text, not bold.
  • Press "Add an item" and then a checkbox right away. The field closes and the box is ticked.
  • Delete every item. "Nothing here yet" comes back.
  • Open the developer tools, Application (or Storage) tab, Local Storage. You can see the saved JSON.

<body>
  <h1>Checklist</h1>

  <div class="card">
    <p class="label">Checklist</p>

    <button type="button" id="add">+ Add an item</button>

    <form id="form" hidden>
      <input type="text" id="new-item" maxlength="200" placeholder="What needs doing?" aria-label="New item" autocomplete="off">
    </form>

    <ul id="items"></ul>

    <p id="empty">Nothing here yet</p>

    <div id="footer" hidden>
      <span id="count"></span>
      <button type="button" id="clear">Clear done</button>
    </div>
  </div>

  <script>
    /*
     * A checklist that remembers its items.
     *
     * State:
     *   items   a list of items, e.g. [{ text: "Buy milk", done: false }]
     *
     * Every change saves the list to localStorage and draws it again,
     * so the list is still there after a reload.
     */

    const STORAGE_KEY = "checklist-items";

    // --- State ---------------------------------------------------------------

    let items = load();

    // --- Elements on the page ------------------------------------------------

    const addButton = document.getElementById("add");
    const form = document.getElementById("form");
    const input = document.getElementById("new-item");
    const list = document.getElementById("items");
    const empty = document.getElementById("empty");
    const footer = document.getElementById("footer");
    const count = document.getElementById("count");
    const clearButton = document.getElementById("clear");

    // --- Saving --------------------------------------------------------------

    // localStorage only stores text, so the list is turned into JSON text.
    function save() {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    }

    // Read the saved list. Nothing saved yet, or broken text, gives [].
    function load() {
        try {
            const saved = JSON.parse(localStorage.getItem(STORAGE_KEY));
            return Array.isArray(saved) ? saved : [];
        } catch (error) {
            return [];
        }
    }

    // --- Drawing -------------------------------------------------------------

    // Make the <li> for one item.
    function makeItem(item, index) {
        const li = document.createElement("li");
        if (item.done) {
            li.classList.add("done");
        }

        const box = document.createElement("input");
        box.type = "checkbox";
        box.checked = item.done;
        box.addEventListener("change", function () { toggle(index); });

        // textContent, not innerHTML, so "<b>" typed by the user stays text.
        const text = document.createElement("span");
        text.textContent = item.text;

        const label = document.createElement("label");
        label.append(box, text);

        const deleteButton = document.createElement("button");
        deleteButton.type = "button";
        deleteButton.className = "delete";
        deleteButton.textContent = "×";
        deleteButton.setAttribute("aria-label", "Delete " + item.text);
        deleteButton.addEventListener("click", function () { remove(index); });

        li.append(label, deleteButton);
        return li;
    }

    // The "Nothing here yet" text shows when the list is empty and nobody types.
    function showEmpty() {
        empty.hidden = items.length > 0 || !form.hidden;
    }

    // Draw the whole list from the state.
    function render() {
        list.textContent = "";   // remove the old items
        items.forEach(function (item, index) {
            list.append(makeItem(item, index));
        });

        const done = items.filter(function (item) { return item.done; }).length;
        count.textContent = done + " of " + items.length + " done";
        clearButton.disabled = done === 0;
        footer.hidden = items.length === 0;
        showEmpty();
    }

    // --- Actions -------------------------------------------------------------

    function add(text) {
        items.push({ text: text, done: false });
        save();
        render();
    }

    function toggle(index) {
        items[index].done = !items[index].done;
        save();
        render();
    }

    function remove(index) {
        items.splice(index, 1);   // remove 1 item at index
        save();
        render();
    }

    function clearDone() {
        items = items.filter(function (item) { return !item.done; });
        save();
        render();
    }

    function openForm() {
        addButton.hidden = true;
        form.hidden = false;
        input.focus();
        showEmpty();
    }

    function closeForm() {
        input.value = "";
        form.hidden = true;
        addButton.hidden = false;
        showEmpty();
    }

    // --- Connect the buttons -------------------------------------------------

    addButton.addEventListener("click", openForm);

    // Enter in the text field submits the form.
    form.addEventListener("submit", function (event) {
        event.preventDefault();   // don't reload the page
        const text = input.value.trim();
        if (text !== "") {
            add(text);
            input.value = "";     // ready for the next item
        }
    });

    input.addEventListener("keydown", function (event) {
        if (event.key === "Escape") {
            closeForm();
        }
    });

    // Clicking somewhere else with an empty field closes it.
    input.addEventListener("blur", function () {
        if (input.value.trim() === "") {
            closeForm();
        }
    });

    clearButton.addEventListener("click", clearDone);

    render();
  </script>
</body>
</html>
