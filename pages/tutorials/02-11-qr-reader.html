QR Reader

Point the camera at a QR code, and the page reads it: a link, a Wi-Fi network with its password, or just text. It can also read a QR code from a picture, and it remembers the last ten codes.

Read the Guitar tuner first: the camera is asked for in the same way as the microphone. The new ideas are using a feature the browser may or may not have, with a library to fall back on, and handling text from strangers safely.

Open the finished reader (it needs a camera, and a page on https)

Step 2: The camera

async function start() {
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        message.textContent = "This browser can't use a camera here. It needs a page on https.";
        return;
    }
    try {
        // "environment" is the camera on the back, if there is one.
        stream = await navigator.mediaDevices.getUserMedia({
            video: { facingMode: "environment" },
            audio: false
        });
    } catch (error) {
        message.textContent = error.name === "NotFoundError"
            ? "No camera found. You can still read a QR code from a picture."
            : "The page may not use the camera. Allow it in the browser's settings.";
        return;
    }
    video.srcObject = stream;
    await video.play();
    scanning = true;
    viewfinder.classList.remove("found");
    startButton.textContent = "Stop";
    startButton.className = "btn btn-warning";
    message.textContent = "Point the camera at a QR code.";
    requestAnimationFrame(scan);
}
  • getUserMedia asks for the camera with video, the way the Guitar tuner asks for the microphone with audio. It only works on https:// pages.
  • facingMode: "environment" asks for the camera on the back, the one facing the world. A computer only has one camera, and gets that.
  • The camera goes into the video with video.srcObject = stream, and play() starts it.
  • On an iPhone, a video jumps to full screen when it plays, unless it has playsinline. That is why the <video> tag has it. muted lets it play without a tap.

Stopping turns the camera off, and its light with it. It is also called when you switch to another app, through the visibilitychange event, so the camera never stays on in the background:

function stop(text) {
    scanning = false;
    if (stream !== null) {
        stream.getTracks().forEach(function (track) { track.stop(); });   // the camera light goes off
        stream = null;
    }
    video.srcObject = null;
    startButton.textContent = current === null ? "Start" : "Scan again";
    startButton.className = "btn btn-success";
    message.textContent = text;
}

Step 3: Two ways to read a code

Reading a QR code is hard. The reader must find the three squares in the corners, straighten the picture if the code is held at an angle, read the black and white squares as bits, and fix the bits that were misread. Some browsers have all of that built in: the BarcodeDetector. But only some: Chrome and Edge on Android and Mac have it, Safari on iPhone and Firefox don't.

So we first check if the browser has one that reads QR codes. Checking for a feature, instead of for a browser name, is called feature detection:

async function makeDetector() {
    if (!("BarcodeDetector" in window)) {
        return null;
    }
    try {
        const formats = await BarcodeDetector.getSupportedFormats();
        if (formats.includes("qr_code")) {
            return new BarcodeDetector({ formats: ["qr_code"] });
        }
    } catch (error) {
        // No built-in reader after all.
    }
    return null;
}

For all other browsers we use jsQR, a library: someone else's code, in a file of its own, that we load before our own script. Pin the version, here 1.4.0, so it can't change under you, and add integrity, a fingerprint of the file. If the file on the other site is ever changed, the fingerprint doesn't match and the browser won't run it:

<script src="https://cdn.jsdelivr.net/npm/jsqr@1.4.0/dist/jsQR.js"
        integrity="sha384-b5Ya4Bq3qCyz39m2ISh+4DxjAIljdeFwK/BsXLuj9gugaNwAcj/ia15fxNZL9Nlx"
        crossorigin="anonymous"></script>

The QR Reader on this site keeps its own copy of the file instead, next to its other files, so it also works offline when it is installed as an app.

Then one function reads a code either way:

async function readCode(picture, width, height, thorough) {
    if (detector !== null) {
        try {
            const codes = await detector.detect(picture);
            return codes.length > 0 ? codes[0].rawValue : null;
        } catch (error) {
            return null;
        }
    }
    const scale = Math.min(1, (thorough ? PICTURE_SIZE : VIDEO_SIZE) / Math.max(width, height));
    canvas.width = Math.round(width * scale);
    canvas.height = Math.round(height * scale);
    ctx.drawImage(picture, 0, 0, canvas.width, canvas.height);
    const pixels = ctx.getImageData(0, 0, canvas.width, canvas.height);
    const code = jsQR(pixels.data, pixels.width, pixels.height, {
        inversionAttempts: thorough ? "attemptBoth" : "dontInvert"
    });
    return code ? code.data : null;
}
  • The BarcodeDetector takes the video or picture as it is, and gives a list of the codes it found.
  • jsQR needs the pixels. We draw the picture on a canvas that is never shown, and getImageData gives its pixels as a long list of numbers: red, green, blue and alpha, for every pixel.
  • A camera picture can be 1920 pixels wide, and jsQR is slow on that many pixels. Shrunk to at most 640, it is much faster, and still sharp enough for a code that fills the frame.
  • Most codes are dark on light. inversionAttempts: "dontInvert" only looks for those, which is twice as fast. For a picture, where speed doesn't matter, "attemptBoth" also finds light codes on a dark background.

Step 4: Looking a few times a second

function scan(now) {
    if (!scanning) {
        return;
    }
    requestAnimationFrame(scan);
    if (busy || now - lastScan < SCAN_EVERY || video.readyState < video.HAVE_CURRENT_DATA) {
        return;
    }
    busy = true;
    lastScan = now;
    readCode(video, video.videoWidth, video.videoHeight, false).then(function (text) {
        busy = false;
        if (text !== null && scanning) {
            viewfinder.classList.add("found");
            if (navigator.vibrate) {
                navigator.vibrate(80);
            }
            show(text);
            remember(text);
            stop("Found a QR code.");
        }
    });
}

This runs every frame, about 60 times a second, but only looks for a code every 150 milliseconds: that is plenty, and leaves the phone time for everything else. readCode takes a little while, so busy makes sure we never start a new look before the last one is done. When a code is found, the frame turns green, the phone buzzes, and the camera stops.

Step 5: What is in the code?

A QR code just holds text. Some text follows a pattern: a link starts with https://, and a Wi-Fi network looks like WIFI:T:WPA;S:MyNetwork;P:secret;;, where S is the network's name and P the password.

function describe(text) {
    if (/^https?:\/\//i.test(text)) {
        let host = "";
        try {
            host = new URL(text).hostname;
        } catch (error) {
            // Not a real address after all: show it as text.
            return { type: "Text", text: text, copy: text };
        }
        return { type: "Link", text: text, copy: text, link: text, host: host };
    }
    if (/^WIFI:/i.test(text)) {
        // WIFI:T:WPA;S:name;P:password;; where \ makes the next letter plain.
        const fields = {};
        const pattern = /([A-Z]):((?:\\.|[^;])*);/g;
        let match;
        while ((match = pattern.exec(text.slice(5))) !== null) {
            fields[match[1]] = match[2].replace(/\\(.)/g, "$1");
        }
        if (fields.S) {
            return {
                type: "Wi-Fi network",
                text: "Network: " + fields.S + (fields.P ? "\nPassword: " + fields.P : ""),
                copy: fields.P || fields.S
            };
        }
    }
    return { type: "Text", text: text, copy: text };
}
  • new URL(text) takes a link apart. Its hostname is the site it goes to, like www.anderswik.se.
  • In the Wi-Fi pattern, a ; ends each field. A name or password that has a ; in it writes \; instead. The regular expression ((?:\\.|[^;])*) reads "any number of: a backslash and any letter, or any letter but ;". Then replace(/\\(.)/g, "$1") takes the backslashes away.

Step 6: Showing it safely

A QR code on a poster could have been put there by anyone. It may hold a link to a fake bank, or text made to look like HTML. Three rules keep the reader safe:

function show(text) {
    current = describe(text);
    resultType.textContent = current.type;
    resultText.textContent = current.text;   // textContent, never innerHTML: a code may hold anything
    resultHost.hidden = !current.host;
    resultHost.textContent = current.host ? "Goes to " + current.host : "";
    openLink.hidden = !current.link;
    openLink.href = current.link || "#";
    copyButton.textContent = current.type === "Wi-Fi network" ? "Copy password" : "Copy";
    resultBox.hidden = false;
}
  • Text, never HTML. textContent shows the code as letters. If we used innerHTML, a code with <img src=x onerror=...> in it would run its own script on our page.
  • Never open a link by itself. You see the whole address, and the site it goes to, and decide. A code for https://www.anderswik.se.bad-site.com shows "Goes to www.anderswik.se.bad-site.com".
  • Only web links get a button. Only http and https links can be opened. A link like javascript:... is shown as text. The link opens in a new tab with rel="noopener", so the new page can't reach back to ours.

Copy uses the clipboard, which also only works on https pages:

copyButton.addEventListener("click", async function () {
    if (current === null) {
        return;
    }
    const label = copyButton.textContent;
    try {
        await navigator.clipboard.writeText(current.copy);
        copyButton.textContent = "Copied";
    } catch (error) {
        copyButton.textContent = "Can't copy here";
    }
    setTimeout(function () { copyButton.textContent = label; }, 1500);
});

Step 7: Pictures, and codes read before

A computer may have no camera, and a QR code may be in a screenshot. The file input lets you pick a picture. createImageBitmap turns the file into a picture that both readers can take:

fileInput.addEventListener("change", async function () {
    const file = fileInput.files[0];
    if (!file) {
        return;
    }
    if (scanning) {
        stop("");
    }
    let picture;
    try {
        picture = await createImageBitmap(file);
    } catch (error) {
        message.textContent = "That file isn't a picture this browser can open.";
        return;
    }
    const text = await readCode(picture, picture.width, picture.height, true);
    fileInput.value = "";
    if (text === null) {
        message.textContent = "No QR code found in the picture. Try a sharper one, or one closer to the code.";
        return;
    }
    show(text);
    remember(text);
    startButton.textContent = "Start";
    message.textContent = "Read a QR code from the picture.";
});

Every code read is saved in localStorage, newest first, at most ten. A code read again moves to the top, instead of being in the list twice:

function remember(text) {
    history = [text].concat(history.filter(function (old) { return old !== text; })).slice(0, HISTORY_LENGTH);
    try {
        localStorage.setItem(HISTORY_KEY, JSON.stringify(history));
    } catch (error) {
        // Not saved, but the list still works until the page is closed.
    }
    drawHistory();
}

The list is made with createElement and textContent, for the same reason as in step 6.

Step 8: Test it

To test reading without holding up a phone, make a few QR codes as pictures, for example with Python's qrcode package, and give them to readCode. Tested like that in Firefox, which has no BarcodeDetector and so uses jsQR:

  • A link, a line of text, and a Wi-Fi code with a \; in both the name and the password were all read right, and describe gave the right name and password.
  • A light code on a dark background was only found with "attemptBoth", as expected: so pictures find it, and the camera, which is quicker, doesn't.

Then test with the camera:

  • Read a QR code from a phone screen, a poster, or a package. The frame should turn green.
  • Read a Wi-Fi code, like the one on many routers. Copy password should copy just the password.
  • Read the same code twice. It should only be in the list once.
  • Switch to another app while it scans. The camera light should go off.
  • The line at the bottom tells which reader the browser uses. Try a phone with Chrome and one with Safari.

        const label = copyButton.textContent;
        try {
            await navigator.clipboard.writeText(current.copy);
            copyButton.textContent = "Copied";
        } catch (error) {
            copyButton.textContent = "Can't copy here";
        }
        setTimeout(function () { copyButton.textContent = label; }, 1500);
    });

    // --- Reading a picture ---------------------------------------------------------

    fileInput.addEventListener("change", async function () {
        const file = fileInput.files[0];
        if (!file) {
            return;
        }
        if (scanning) {
            stop("");
        }
        let picture;
        try {
            picture = await createImageBitmap(file);
        } catch (error) {
            message.textContent = "That file isn't a picture this browser can open.";
            return;
        }
        const text = await readCode(picture, picture.width, picture.height, true);
        fileInput.value = "";
        if (text === null) {
            message.textContent = "No QR code found in the picture. Try a sharper one, or one closer to the code.";
            return;
        }
        show(text);
        remember(text);
        startButton.textContent = "Start";
        message.textContent = "Read a QR code from the picture.";
    });

    // --- Codes read before -----------------------------------------------------------

    function loadHistory() {
        try {
            const saved = JSON.parse(localStorage.getItem(HISTORY_KEY));
            return Array.isArray(saved) ? saved : [];
        } catch (error) {
            return [];
        }
    }

    // Put a code first in the list, without doubles, and keep the newest few.
    function remember(text) {
        history = [text].concat(history.filter(function (old) { return old !== text; })).slice(0, HISTORY_LENGTH);
        try {
            localStorage.setItem(HISTORY_KEY, JSON.stringify(history));
        } catch (error) {
            // Not saved, but the list still works until the page is closed.
        }
        drawHistory();
    }

    function drawHistory() {
        historyList.textContent = "";
        history.forEach(function (text) {
            const item = document.createElement("li");
            const button = document.createElement("button");
            button.type = "button";
            button.textContent = text;
            button.addEventListener("click", function () { show(text); });
            item.appendChild(button);
            historyList.appendChild(item);
        });
        historyBox.hidden = history.length === 0;
    }

    clearButton.addEventListener("click", function () {
        history = [];
        try {
            localStorage.removeItem(HISTORY_KEY);
        } catch (error) {
            // Nothing saved to remove.
        }
        drawHistory();
    });

    // --- Start -----------------------------------------------------------------------

    startButton.addEventListener("click", function () {
        if (scanning) {
            stop("Press Start to read a QR code.");
        } else {
            start();
        }
    });

    // Switch the camera off when the page is hidden, like when you change apps.
    document.addEventListener("visibilitychange", function () {
        if (document.hidden && scanning) {
            stop("Press Start to read a QR code.");
        }
    });

    makeDetector().then(function (found) {
        detector = found;
        engineText.textContent = detector !== null
            ? "Reading QR codes with the browser's own BarcodeDetector."
            : "Reading QR codes with jsQR.";
    });
    drawHistory();
  </script>
</body>
