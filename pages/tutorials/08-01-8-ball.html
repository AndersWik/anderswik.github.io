Magic 8 ball

This chapter builds games. We start with the smallest: a magic 8 ball. You ask a question, the ball shakes, and you get an answer. The answer is random, but random numbers are the heart of many games, and the ball is a good place to learn them.

Open the finished game

If you haven't built a page from scratch before, read Building a utility page first.

Step 1: The page

Create 8-ball.html from the skeleton.

<h1>Magic 8 ball</h1>

<div class="ball" id="eightball"><span>8</span></div>

<form id="ask-form" autocomplete="off">
  <input type="text" id="question" placeholder="Ask your question..." aria-label="Your question">
  <button type="submit">Shake the ball</button>
</form>

<p id="result" aria-live="polite"></p>

The question box and the button are in a <form>, so pressing Enter works as well as clicking. aria-live="polite" makes screen readers read out the answer when it changes.

Step 2: A ball made with CSS

We don't need a picture of the ball. CSS can draw it:

body {
    font-family: sans-serif;
    max-width: 800px;
    margin: 40px auto;
    padding: 0 16px;
}

body {
    text-align: center;
}

.ball {
    width: 200px;
    height: 200px;
    margin: 0 auto 20px;
    border-radius: 50%;
    background: radial-gradient(circle at 35% 30%, #555555, #000000 60%);
    display: flex;
    align-items: center;
    justify-content: center;
}

.ball span {
    width: 80px;
    height: 80px;
    border-radius: 50%;
    background-color: white;
    font-size: 48px;
    font-weight: bold;
    line-height: 80px;
}

.shake-ball {
    animation: shake 0.5s infinite;
}

@keyframes shake {
    0% { transform: translate(0, 0) rotate(0deg); }
    25% { transform: translate(-4px, 2px) rotate(-3deg); }
    50% { transform: translate(4px, -2px) rotate(3deg); }
    75% { transform: translate(-2px, -2px) rotate(-2deg); }
    100% { transform: translate(0, 0) rotate(0deg); }
}

input,
button {
    padding: 10px;
    font-size: 18px;
}

#result {
    min-height: 1.5em;
    font-size: 24px;
}
  • border-radius: 50% makes a square into a circle.
  • radial-gradient(circle at 35% 30%, ...) paints a light spot up and to the left that fades to black. It makes the ball look round and shiny.
  • The white circle with the 8 is a <span> inside the ball, centered with flexbox.
  • @keyframes shake describes a small wobble: move a few pixels and turn a few degrees, then back. The class shake-ball plays it over and over (infinite), every half second. JavaScript only has to add and remove the class.

Step 4: Picking at random

Math.random() gives a random number from 0 up to, but not including, 1, like 0.7341. To pick an item from a list:

function randomItem(list) {
    return list[Math.floor(Math.random() * list.length)];
}

With 10 answers, Math.random() * 10 is between 0 and 9.999..., and Math.floor rounds down to a whole number from 0 to 9. That is exactly the range of index numbers in the list, and every answer is equally likely. You will use this little function in almost every game.

Step 5: Shake, wait, answer

function shake(event) {
    event.preventDefault();

    if (questionInput.value.trim() === "") {
        resultText.textContent = "Please ask a question first!";
        return;
    }

    if (ball.classList.contains("shake-ball")) {
        return;
    }

    resultText.textContent = "";
    ball.classList.add("shake-ball");

    setTimeout(function () {
        ball.classList.remove("shake-ball");
        resultText.textContent = randomItem(ANSWERS);
    }, SHAKE_TIME);
}

form.addEventListener("submit", shake);
  • event.preventDefault() stops the form from reloading the page.
  • trim() removes spaces, so a question of only spaces counts as empty.
  • If the ball is already shaking, we ignore the click. Otherwise, clicking three times would give three answers, one after the other.
  • setTimeout(function, 500) runs the function once, after 500 milliseconds. Until then the ball shakes, which builds a little suspense.

Step 6: Test it

  • Press the button with an empty box, and with only spaces.
  • Ask a question and press Enter instead of clicking.
  • Click many times quickly. You should still get only one answer per shake.

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <title>Magic 8 ball</title>
  <style>
    body {
        font-family: sans-serif;
        max-width: 800px;
        margin: 40px auto;
        padding: 0 16px;
    }

    body {
        text-align: center;
    }

    .ball {
        width: 200px;
        height: 200px;
        margin: 0 auto 20px;
        border-radius: 50%;
        background: radial-gradient(circle at 35% 30%, #555555, #000000 60%);
        display: flex;
        align-items: center;
        justify-content: center;
    }

    .ball span {
        width: 80px;
        height: 80px;
        border-radius: 50%;
        background-color: white;
        font-size: 48px;
        font-weight: bold;
        line-height: 80px;
    }

    .shake-ball {
        animation: shake 0.5s infinite;
    }

    @keyframes shake {
        0% { transform: translate(0, 0) rotate(0deg); }
        25% { transform: translate(-4px, 2px) rotate(-3deg); }
        50% { transform: translate(4px, -2px) rotate(3deg); }
        75% { transform: translate(-2px, -2px) rotate(-2deg); }
        100% { transform: translate(0, 0) rotate(0deg); }
    }

    input,
    button {
        padding: 10px;
        font-size: 18px;
    }

    #result {
        min-height: 1.5em;
        font-size: 24px;
    }
  </style>
</head>
<body>
  <h1>Magic 8 ball</h1>

  <div class="ball" id="eightball"><span>8</span></div>

  <form id="ask-form" autocomplete="off">
    <input type="text" id="question" placeholder="Ask your question..." aria-label="Your question">
    <button type="submit">Shake the ball</button>
  </form>

  <p id="result" aria-live="polite"></p>

  <script>
    /*
     * Magic 8 ball: ask a question, shake the ball, get a random answer.
     *
     * The question is not used for anything; the answer is picked at random.
     * The shaking is a CSS animation that we turn on by adding a class.
     */

    // --- Elements on the page ------------------------------------------------

    const ball = document.getElementById("eightball");
    const form = document.getElementById("ask-form");
    const questionInput = document.getElementById("question");
    const resultText = document.getElementById("result");

    // --- Data ----------------------------------------------------------------

    const ANSWERS = [
        "Yes, definitely.",
        "It is certain.",
        "Most likely.",
        "Ask again later.",
        "Cannot predict right now.",
        "Don’t count on it.",
        "Very doubtful.",
        "No way.",
        "Probably not.",
        "Maybe."
    ];

    const SHAKE_TIME = 500; // milliseconds, the same as the animation in the CSS

    // --- Helpers -------------------------------------------------------------

    // Math.random() gives a number from 0 up to (not including) 1.
    // Times the length, rounded down, gives an index from 0 to length - 1.
    function randomItem(list) {
        return list[Math.floor(Math.random() * list.length)];
    }

    // --- Actions -------------------------------------------------------------

    function shake(event) {
        // The form would reload the page; we handle it here instead.
        event.preventDefault();

        if (questionInput.value.trim() === "") {
            resultText.textContent = "Please ask a question first!";
            return;
        }

        // Already shaking? Then wait for that answer.
        if (ball.classList.contains("shake-ball")) {
            return;
        }

        resultText.textContent = "";
        ball.classList.add("shake-ball");

        setTimeout(function () {
            ball.classList.remove("shake-ball");
            resultText.textContent = randomItem(ANSWERS);
        }, SHAKE_TIME);
    }

    // --- Connect the page ----------------------------------------------------

    // A form lets the user press Enter as well as click the button.
    form.addEventListener("submit", shake);
  </script>
</body>
</html>
