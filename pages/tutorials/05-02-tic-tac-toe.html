Tic Tac Toe

X and O take turns placing their mark on a 3 by 3 board. Three in a row, across, down or diagonally, wins. Play against a friend, or against the computer, which plays O. The easy computer picks a random square. The smart one follows a few rules that make it hard to beat.

This is the smallest board game in the tutorials, so it is a good place to see the whole pattern once more: a list holds the board, the page only shows it, and the computer uses the same play as a click.

Step 2: The board and its lines

const LINES = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8],
    [0, 3, 6], [1, 4, 7], [2, 5, 8],
    [0, 4, 8], [2, 4, 6]
];

let board = [];        // "", "X" or "O" for each square

The board is one list of nine squares, numbered 0 to 8 row by row. A board this small has only eight lines of three, so we just write them all down, like the mills in Kvarn.

function winningLine(who) {
    return LINES.find(function (line) {
        return line.every(function (index) {
            return board[index] === who;
        });
    });
}

Find a line where every square has who's mark. find gives the line, or undefined if there is none. The line is useful: it tells us which squares to color green.

Same order as in Connect Four: change the board, check for a win, check for a full board, and only then switch player. And as in Connect Four vs computer, clicks are ignored on the computer's turn, and newGame cancels a computer move that is still waiting.

The squares get their click listeners in a loop. forEach gives each button its index, so each one knows its own square:

squares.forEach(function (square, index) {
    square.addEventListener("click", function () {

    });
});

Step 4: The smart computer

First, a helper: is there a line where who has two marks and the third square is free?

function winningSquare(who) {
    for (const line of LINES) {
        const marks = line.map(function (index) {
            return board[index];
        });
        const mine = marks.filter(function (mark) {
            return mark === who;
        }).length;
        if (mine === 2 && marks.includes("")) {
            return line[marks.indexOf("")];
        }
    }
    return undefined;
}

marks is what is in the line's three squares, like ["O", "", "O"]. If two of them are who's and one is empty, the empty one wins. marks.indexOf("") is where in the line it is, and line[...] turns that into the square's number on the board.

function smartMove() {
    const win = winningSquare("O");
    if (win !== undefined) {
        return win;
    }
    const block = winningSquare("X");
    if (block !== undefined) {
        return block;
    }
    if (board[4] === "") {
        return 4;
    }
    const corners = [0, 2, 6, 8].filter(function (index) {
        return board[index] === "";
    });
    if (corners.length > 0) {
        return randomItem(corners);
    }
    return randomItem(emptySquares());
}

Win if possible, block if needed, then take the middle, which is part of four lines, then a corner, part of three. These are the same rules as the easy computer in Connect Four vs computer. They are enough to make Tic Tac Toe hard to win, but not impossible: if you threaten two lines at once (a fork), it can only block one.

In the first version, the middle and corner rules used a variable called board that didn't exist. As soon as the computer had nothing to win or block, it crashed, and since "smart" was the default, the game stopped after a couple of moves.

Step 5: Test it

  • Win across, down and diagonally. The three squares should turn green.
  • Fill the board without a winner. It should say it's a draw.
  • Against the smart computer, start in a corner and see that it takes the middle.
  • Make two threats at once and win.

  <script>
    /*
     * Tic Tac Toe: X and O take turns. Three in a row, across, down or
     * diagonally, wins. Play against a friend or the computer, which plays O.
     *
     *   Easy:  picks a random free square.
     *   Smart: wins if it can, blocks you if you are about to win, and
     *          otherwise likes the middle, then the corners.
     */

    // --- Settings ------------------------------------------------------------

    // The squares are numbered like this:   0 1 2
    //                                       3 4 5
    //                                       6 7 8
    const LINES = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8],   // across
        [0, 3, 6], [1, 4, 7], [2, 5, 8],   // down
        [0, 4, 8], [2, 4, 6]               // diagonal
    ];
    const COMPUTER_DELAY = 500;   // ms before the computer moves

    // --- Elements on the page ------------------------------------------------

    const squares = document.querySelectorAll(".square");
    const statusText = document.getElementById("status");
    const restartButton = document.getElementById("reset");
    const modeSelect = document.getElementById("mode");

    // --- State ---------------------------------------------------------------

    let board = [];        // "", "X" or "O" for each square
    let player = "X";
    let gameOver = false;
    let computerTimer = null;

    // --- Rules ---------------------------------------------------------------

    // The line that "who" has three in a row on, or undefined.
    function winningLine(who) {
        return LINES.find(function (line) {
            return line.every(function (index) {
                return board[index] === who;
            });
        });
    }

    function emptySquares() {
        const empty = [];
        board.forEach(function (mark, index) {
            if (mark === "") {
                empty.push(index);
            }
        });
        return empty;
    }

    function randomItem(list) {
        return list[Math.floor(Math.random() * list.length)];
    }

    // --- Playing -------------------------------------------------------------

    function isComputersTurn() {
        return modeSelect.value !== "pvp" && player === "O";
    }

    function play(index) {
        if (gameOver || board[index] !== "") {
            return;
        }
        board[index] = player;
        squares[index].textContent = player;

        const line = winningLine(player);
        if (line) {
            line.forEach(function (i) {
                squares[i].classList.add("winning");
            });
            gameOver = true;
            if (modeSelect.value === "pvp") {
                statusText.textContent = player + " wins!";
            } else {
                statusText.textContent = player === "X" ? "You win!" : "The computer wins!";
            }
        } else if (emptySquares().length === 0) {
            gameOver = true;
            statusText.textContent = "It's a draw!";
        } else {
            player = player === "X" ? "O" : "X";
            statusText.textContent = isComputersTurn() ? "The computer is thinking..." : player + "'s turn";
            if (isComputersTurn()) {
                computerTimer = setTimeout(computerMove, COMPUTER_DELAY);
            }
        }
    }

    // --- The computer --------------------------------------------------------

    // A free square that would give "who" three in a row, or undefined.
    function winningSquare(who) {
        for (const line of LINES) {
            const marks = line.map(function (index) {
                return board[index];
            });
            const mine = marks.filter(function (mark) {
                return mark === who;
            }).length;
            if (mine === 2 && marks.includes("")) {
                return line[marks.indexOf("")];
            }
        }
        return undefined;
    }

    function smartMove() {
        const win = winningSquare("O");
        if (win !== undefined) {
            return win;
        }
        const block = winningSquare("X");
        if (block !== undefined) {
            return block;
        }
        if (board[4] === "") {
            return 4;
        }
        const corners = [0, 2, 6, 8].filter(function (index) {
            return board[index] === "";
        });
        if (corners.length > 0) {
            return randomItem(corners);
        }
        return randomItem(emptySquares());
    }

    function computerMove() {
        const index = modeSelect.value === "smart" ? smartMove() : randomItem(emptySquares());
        play(index);
    }

    // --- Starting ------------------------------------------------------------

    function newGame() {
        clearTimeout(computerTimer);
        board = ["", "", "", "", "", "", "", "", ""];
        squares.forEach(function (square) {
            square.textContent = "";
            square.classList.remove("winning");
        });
        player = "X";
        gameOver = false;
        statusText.textContent = "X's turn";
    }

    // --- Connect the page ----------------------------------------------------

    squares.forEach(function (square, index) {
        square.addEventListener("click", function () {
            if (!isComputersTurn()) {
                play(index);
            }
        });
    });
    restartButton.addEventListener("click", newGame);
    modeSelect.addEventListener("change", newGame);

    newGame();
  </script>
</body>
</html>
