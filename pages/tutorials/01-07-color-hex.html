Color picker

We build a color picker. Sliders for red, green, blue and alpha show the color as rgb(), rgba() and hex, together with its opposite color, 10 lighter and 10 darker versions, and a list of favorites. The color is also kept in the address, like color-hex.html#ff8800, so you can bookmark or share it.

Open the finished color picker

How screen colors work

A pixel mixes red, green and blue light, each from 0 to 255. The same color can be written in different ways:

rgb(255, 136, 0)          three numbers
#ff8800                   the same numbers in hexadecimal: ff = 255, 88 = 136, 00 = 0
rgba(255, 136, 0, 0.5)    plus alpha: 0 is see-through, 1 is solid

Hexadecimal counts with 16 digits, 0-9 and then a-f, so two hex digits hold 0 to 255. In JavaScript, (255).toString(16) gives "ff" and parseInt("ff", 16) gives 255.

The plan

The sliders are the state. Every time one moves, render() reads all of them and redraws everything. There is one way to draw the page, so it can't get out of step with the sliders.

Step 2: Colors as objects

We store a color as an object like { red: 255, green: 136, blue: 0 }. The helpers below only calculate. They never touch the page, so you can try every one in the console.

// Keep a value inside 0-255 and make it a whole number.
function clamp(value) {
    return Math.min(255, Math.max(0, Math.round(value)));
}

// { red: 255, green: 136, blue: 0 } -> "#ff8800"
function toHex(color) {
    const parts = [color.red, color.green, color.blue].map(function (value) {
        return clamp(value).toString(16).padStart(2, "0");
    });
    return "#" + parts.join("");
}

padStart(2, "0") is needed because (8).toString(16) is only "8", and a hex color needs "08".

Going the other way, a regular expression checks that the text is exactly six hex digits. The i at the end means upper and lower case both count. slice(0, 2) takes the first two characters.

// "#ff8800" -> { red: 255, green: 136, blue: 0 }, or null if not valid.
function fromHex(hex) {
    hex = hex.replace("#", "");

    if (!/^[0-9a-f]{6}$/i.test(hex)) {
        return null;
    }
    return {
        red: parseInt(hex.slice(0, 2), 16),
        green: parseInt(hex.slice(2, 4), 16),
        blue: parseInt(hex.slice(4, 6), 16)
    };
}

Step 3: Alpha, opposite, lighter and darker

A see-through color on a white page looks like a lighter solid color. We work out that solid color by mixing each part with white (255), using alpha as the weight:

function onWhite(color, alpha) {
    function mix(value) {
        return clamp(alpha * value + (1 - alpha) * 255);
    }
    return { red: mix(color.red), green: mix(color.green), blue: mix(color.blue) };
}

// The opposite color: black <-> white, red <-> cyan...
function invert(color) {
    return { red: 255 - color.red, green: 255 - color.green, blue: 255 - color.blue };
}

// Positive amount = lighter, negative = darker.
function shade(color, amount) {
    return {
        red: clamp(color.red + amount),
        green: clamp(color.green + amount),
        blue: clamp(color.blue + amount)
    };
}

The opposite color is used for the text, so it can always be read on the background.

Step 4: Swatches built with createElement

Each swatch is a link like <a href="#ff8800">. We build it with document.createElement and set textContent, instead of joining HTML strings. That way there is no risk of broken HTML.

function showSwatches(list, hexColors) {
    list.innerHTML = "";    // remove the old swatches

    hexColors.forEach(function (hex) {
        const link = document.createElement("a");
        link.href = hex;
        link.className = "go-to-color";
        link.style.backgroundColor = hex;
        link.style.color = toHex(invert(fromHex(hex)));

        const label = document.createElement("span");
        label.textContent = hex;
        link.appendChild(label);

        list.appendChild(link);
    });
}

Step 5: render()

function render() {
    const color = {
        red: Number(redSlider.value),
        green: Number(greenSlider.value),
        blue: Number(blueSlider.value)
    };
    const alpha = Number(alphaSlider.value) / 100;

    const solid = onWhite(color, alpha);
    const solidHex = toHex(solid);
    const invertedHex = toHex(invert(solid));

    const rgb = "rgb(" + solid.red + ", " + solid.green + ", " + solid.blue + ")";
    const rgba = "rgba(" + color.red + ", " + color.green + ", " + color.blue + ", " + alpha + ")";

    paintPanel(rgbPanel, rgbText, rgb, invertedHex);
    paintPanel(rgbaPanel, rgbaText, rgba, invertedHex);
    paintPanel(hexPanel, hexText, solidHex, invertedHex);
    paintPanel(invertedPanel, invertedText, invertedHex, solidHex);

    hexInput.value = toHex(color);
    history.replaceState(null, "", toHex(color));

    const lighter = [];
    const darker = [];
    for (let step = 0; step < 50; step += 5) {
        lighter.push(toHex(shade(color, step)));
        darker.push(toHex(shade(color, -step)));
    }
    showSwatches(lighterList, lighter);
    showSwatches(darkerList, darker);
}

paintPanel() sets a panel's background and text color and writes the CSS text into it. A slider's value is always text, even for type="range", so we use Number().

Step 6: The color in the address

The part of the address after # is the hash. Changing it doesn't load a new page, but the browser fires a "hashchange" event. We use that in two directions:

  • Clicking a swatch, which is a link to #ff8800, changes the hash. We listen for "hashchange" and move the sliders.
  • Moving a slider writes the color into the hash with history.replaceState(). That changes the address without adding a history entry, so the back button doesn't have to step through every slider move, and it doesn't fire "hashchange".

The second part matters more than it looks. Without it, if you click #ff8800, move a slider and click #ff8800 again, the hash doesn't change, no event fires, and nothing happens.

function loadHex(hex) {
    const color = fromHex(hex);

    if (color === null) {
        return;
    }
    redSlider.value = color.red;
    greenSlider.value = color.green;
    blueSlider.value = color.blue;
    alphaSlider.value = 100;
    render();
}

window.addEventListener("hashchange", function () {
    loadHex(window.location.hash);
});

A valid hex typed in the hex box goes into the address too, which loads it the same way:

hexInput.addEventListener("input", function () {
    if (fromHex(hexInput.value) !== null) {
        window.location.hash = hexInput.value.replace("#", "");
    }
});

Step 7: Start

At the start we show the favorites. If the page was opened with a color in the address, we load it, otherwise we draw the sliders' default color.

showSwatches(favoriteList, favorites);

if (fromHex(window.location.hash) !== null) {
    loadHex(window.location.hash);
} else {
    render();
}

Step 8: Test it

  • Drag each slider. All panels and swatches update while you drag, because "input" fires during the drag and "change" only fires when you let go.
  • Type ff8800 in the hex box. The address ends with #ff8800.
  • Click a swatch, move a slider, then click the same swatch again. It should load again.
  • Add #253746 to the end of the address and press Enter in a new tab. It starts with that color.

    /*
     * Color Hex: pick a color with sliders and see it in different formats.
     *
     * A screen color is made of red, green and blue light, each 0-255.
     * The same color can be written in different ways:
     *
     *   rgb(255, 136, 0)          three numbers
     *   #ff8800                   the same three numbers in hexadecimal
     *   rgba(255, 136, 0, 0.5)    plus alpha: how see-through it is (0-1)
     *
     * The sliders on the page are the "state": every time one moves we read
     * them all and redraw everything from scratch (the render function).
     *
     * Lighter/darker/favorite colors are links like <a href="#ff8800">.
     * Clicking one changes the "#..." part of the page address, and we
     * listen for that change to load the color. Moving a slider also writes
     * the color into the address, so the page can be shared or bookmarked.
     */

    // --- Elements on the page ------------------------------------------------

    const redSlider = document.getElementById("red");
    const greenSlider = document.getElementById("green");
    const blueSlider = document.getElementById("blue");
    const alphaSlider = document.getElementById("alpha");
    const hexInput = document.getElementById("hex");

    const rgbPanel = document.getElementById("color-panel-rgb");
    const rgbText = document.getElementById("color-panel-text-rgb");
    const rgbaPanel = document.getElementById("color-panel-rgba");
    const rgbaText = document.getElementById("color-panel-text-rgba");
    const hexPanel = document.getElementById("color-panel-hex");
    const hexText = document.getElementById("color-panel-text-hex");
    const invertedPanel = document.getElementById("color-panel-inverted");
    const invertedText = document.getElementById("color-panel-text-inverted");

    const lighterList = document.getElementById("lighter-color-container");
    const darkerList = document.getElementById("darker-color-container");
    const favoriteList = document.getElementById("favorite-color-container");

    const favorites = ["#afc0bc", "#7c878f", "#253746", "#8db9ca", "#ca9a8e",
                       "#d9c89e", "#899064", "#6ba4b8", "#d9d9d6", "#caffca"];

    // --- Color helpers -------------------------------------------------------
    // A color is an object like { red: 255, green: 136, blue: 0 }.

    // Keep a value inside 0-255 and make it a whole number.
    function clamp(value) {
        return Math.min(255, Math.max(0, Math.round(value)));
    }

    // { red: 255, green: 136, blue: 0 } -> "#ff8800"
    function toHex(color) {
        const parts = [color.red, color.green, color.blue].map(function (value) {
            return clamp(value).toString(16).padStart(2, "0");
        });
        return "#" + parts.join("");
    }

    // "#ff8800" -> { red: 255, green: 136, blue: 0 }, or null if not valid.
    function fromHex(hex) {
        hex = hex.replace("#", "");

        if (!/^[0-9a-f]{6}$/i.test(hex)) {
            return null;
        }
        return {
            red: parseInt(hex.slice(0, 2), 16),
            green: parseInt(hex.slice(2, 4), 16),
            blue: parseInt(hex.slice(4, 6), 16)
        };
    }

    // A see-through color on a white page looks like a lighter solid color.
    // Mix each part with white (255) depending on alpha.
    function onWhite(color, alpha) {
        function mix(value) {
            return clamp(alpha * value + (1 - alpha) * 255);
        }
        return { red: mix(color.red), green: mix(color.green), blue: mix(color.blue) };
    }

    // The opposite color: black <-> white, red <-> cyan...
    function invert(color) {
        return { red: 255 - color.red, green: 255 - color.green, blue: 255 - color.blue };
    }

    // Add the same amount to every part. Positive = lighter, negative = darker.
    function shade(color, amount) {
        return {
            red: clamp(color.red + amount),
            green: clamp(color.green + amount),
            blue: clamp(color.blue + amount)
        };
    }

    // --- Drawing -------------------------------------------------------------

    function paintPanel(panel, text, background, textColor) {
        panel.style.backgroundColor = background;
        panel.style.color = textColor;
        text.textContent = "background-color: " + background + ";";
    }

    // Fill a list with color links, e.g. <a href="#ff8800">#ff8800</a>.
    function showSwatches(list, hexColors) {
        list.innerHTML = "";

        hexColors.forEach(function (hex) {
            const link = document.createElement("a");
            link.href = hex;
            link.className = "go-to-color";
            link.style.backgroundColor = hex;
            link.style.color = toHex(invert(fromHex(hex)));

            const label = document.createElement("span");
            label.textContent = hex;
            link.appendChild(label);

            list.appendChild(link);
        });
    }

    // Read the sliders and redraw the whole page from them.
    function render() {
        const color = {
            red: Number(redSlider.value),
            green: Number(greenSlider.value),
            blue: Number(blueSlider.value)
        };
        const alpha = Number(alphaSlider.value) / 100;

        const solid = onWhite(color, alpha);
        const solidHex = toHex(solid);
        const invertedHex = toHex(invert(solid));

        const rgb = "rgb(" + solid.red + ", " + solid.green + ", " + solid.blue + ")";
        const rgba = "rgba(" + color.red + ", " + color.green + ", " + color.blue + ", " + alpha + ")";

        paintPanel(rgbPanel, rgbText, rgb, invertedHex);
        paintPanel(rgbaPanel, rgbaText, rgba, invertedHex);
        paintPanel(hexPanel, hexText, solidHex, invertedHex);
        paintPanel(invertedPanel, invertedText, invertedHex, solidHex);

        hexInput.value = toHex(color);

        // Put the color in the address without adding a history entry.
        // Without this, clicking the same swatch twice would do nothing the
        // second time, because the address would not change.
        history.replaceState(null, "", toHex(color));

            darker.push(toHex(shade(color, -step)));

        showSwatches(darkerList, darker);

        if (color === null) {
            return;

        redSlider.value = color.red;
        greenSlider.value = color.green;
        blueSlider.value = color.blue;

    blueSlider.addEventListener("input", render);

    if (fromHex(window.location.hash) !== null) {
